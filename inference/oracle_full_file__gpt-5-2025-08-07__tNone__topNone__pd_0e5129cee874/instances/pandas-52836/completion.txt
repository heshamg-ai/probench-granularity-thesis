                # Optimized path for masked EAs: build transposed columns
                # by transposing the underlying data and mask in a vectorized way.
                arrays = list(self._iter_column_arrays())
                if arrays and hasattr(arrays[0], "_data") and hasattr(arrays[0], "_mask"):
