    def _getitem_slice(self: _IndexT, slobj: slice) -> _IndexT:
        """
        Fastpath for __getitem__ when we know we have a slice.
        """
        res = self._data[slobj]
        out = type(self)._simple_new(res, name=self._name)
        if self.is_unique:
            # Propagate uniqueness cache when slicing a unique index
            out.__dict__["is_unique"] = True
        return out
