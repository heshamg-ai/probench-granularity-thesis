        def _positions_from_ranges(jrng: RangeIndex, brng: RangeIndex) -> np.ndarray:
            jlen = len(jrng)
            if jlen == 0:
                return np.empty(0, dtype=np.intp)
            i = np.arange(jlen, dtype=np.int64)
            rel = (np.int64(jrng.start) - np.int64(brng.start)) + i * np.int64(jrng.step)
            # floor division and modulo work with negative steps as desired
            k = rel // np.int64(brng.step)
            valid = (rel % np.int64(brng.step) == 0) & (k >= 0) & (k < len(brng))
            out = np.where(valid, k, -1).astype(np.intp, copy=False)
            return out
