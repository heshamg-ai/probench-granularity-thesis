    def isna(self) -> npt.NDArray[np.bool_]:
        """
        Boolean NumPy array indicating if each value is missing.

        This should return a 1-D array the same length as 'self'.
        """
        # Fast paths for zero-null and all-null arrays to avoid Arrow compute overhead
        length = len(self._data)
        nulls = self._data.null_count
        if nulls == 0:
            return np.zeros(length, dtype=np.bool_)
        if nulls == length and length != 0:
            return np.ones(length, dtype=np.bool_)
        return self._data.is_null().to_numpy()
