        ids_all, _, _ = self.grouper.group_info
        # Preallocate result data
        result_data = np.zeros((ngroups, ncols), dtype="int64")

        for j, (colname, ser) in enumerate(obj.items()):
            val = ser._values
            codes, _ = algorithms.factorize(val, sort=False)

            if len(ids_all) == 0:
                # no rows; leave zeros
                continue

            sorter = np.lexsort((codes, ids_all))
            codes_s = codes[sorter]
            ids_s = ids_all[sorter]

            # group boundaries are where group ids change
            # unique observations are where sorted values change
            idx = np.r_[0, 1 + np.nonzero(ids_s[1:] != ids_s[:-1])[0]]
            inc = np.r_[1, codes_s[1:] != codes_s[:-1]]

            # 1st item of each group is a new unique observation
            mask = codes_s == -1
            if dropna:
                inc[idx] = 1
                inc[mask] = 0
            else:
                inc[mask & np.r_[False, mask[:-1]]] = 0
                inc[idx] = 1

            out = np.add.reduceat(inc, idx).astype("int64", copy=False)
            if len(ids_s):
                # NaN/NaT group exists if the head of ids is -1,
                # so remove it from res and exclude its index from idx
                if ids_s[0] == -1:
                    res = out[1:]
                    idx_mapped = idx[np.flatnonzero(idx)]
                else:
                    res = out
                    idx_mapped = idx
            else:
                res = out[1:]
                idx_mapped = idx

            # we might have duplications among the bins
            if len(res) != ngroups:
                # align to full result index length
                res_full = np.zeros(ngroups, dtype=out.dtype)
                if len(ids_s) > 0:
                    res_full[ids_s[idx_mapped]] = res
                res = res_full
