    def _groupby_op(
        self,
        *,
        how: str,
        has_dropped_na: bool,
        min_count: int,
        ngroups: int,
        ids: npt.NDArray[np.intp],
        **kwargs,
    ):
        if isinstance(self.dtype, StringDtype):
            return super()._groupby_op(
                how=how,
                has_dropped_na=has_dropped_na,
                min_count=min_count,
                ngroups=ngroups,
                ids=ids,
                **kwargs,
            )

        pa_type = self._pa_array.type
        if how in ("max", "min") and (
            pa.types.is_timestamp(pa_type) or pa.types.is_duration(pa_type)
        ):
            # Fast path for temporal types using segmented reduction on integer
            # representations, avoiding conversion to masked arrays.
            nobs = len(self)
            if nobs == 0 or ngroups == 0:
                return type(self)(pa.chunked_array([], type=pa_type))

            ids_arr = np.asarray(ids, dtype=np.intp)
            valid_id_mask = ids_arr >= 0

            if valid_id_mask.any():
                # Cast to int64 for reduction; handle nulls via sentinels
                data_int = self._pa_array.cast(pa.int64()).to_numpy()
                na_mask = self.isna()

                data_int = data_int[valid_id_mask]
                na_mask = na_mask[valid_id_mask]
                ids_valid = ids_arr[valid_id_mask]

                if how == "max":
                    sentinel = np.iinfo(np.int64).min
                    out = np.full(ngroups, sentinel, dtype=np.int64)
                    if data_int.size:
                        temp = data_int.copy()
                        temp[na_mask] = sentinel
                        np.maximum.at(out, ids_valid, temp)
                    group_nulls = out == sentinel
                else:
                    sentinel = np.iinfo(np.int64).max
                    out = np.full(ngroups, sentinel, dtype=np.int64)
                    if data_int.size:
                        temp = data_int.copy()
                        temp[na_mask] = sentinel
                        np.minimum.at(out, ids_valid, temp)
                    group_nulls = out == sentinel

                if min_count > 0:
                    counts = np.zeros(ngroups, dtype=np.int64)
                    notna = ~na_mask
                    if notna.any():
                        np.add.at(counts, ids_valid[notna], 1)
                    group_nulls = group_nulls | (counts < min_count)

                result_pa = pa.array(out, type=pa_type, mask=group_nulls, from_pandas=False)
                return type(self)(result_pa)
            else:
                # No valid groups present
                result_pa = pa.array([None] * ngroups, type=pa_type)
                return type(self)(result_pa)

        masked = self._to_masked()

        result = masked._groupby_op(
            how=how,
            has_dropped_na=has_dropped_na,
            min_count=min_count,
            ngroups=ngroups,
            ids=ids,
            **kwargs,
        )
        if isinstance(result, np.ndarray):
            return result
        return type(self)._from_sequence(result, copy=False)
