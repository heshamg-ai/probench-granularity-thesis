    if columns is not None:
        # Fast path: align dict-like data to provided columns without constructing a Series
        columns_list = list(columns)

        # Determine index if not provided, excluding entries that are "missing"
        # (i.e., key absent or scalar NA value)
        if index is None:
            present_vals = []
            missing_mask: list[bool] = []
            for col in columns_list:
                if col in data:
                    val = data[col]
                    miss = is_scalar_na(val)
                    missing_mask.append(miss)
                    if not miss:
                        present_vals.append(val)
                else:
                    missing_mask.append(True)
            # GH10856: raise ValueError if only scalars in dict
            index = _extract_index(present_vals)
        else:
            index = ensure_index(index)
            missing_mask = [
                (not (col in data)) or is_scalar_na(data[col]) for col in columns_list
            ]

        arrays = []
        fill_missing = any(missing_mask) and not is_integer_dtype(dtype)

        # Prepare shared NaN array for missing when dtype is None and copy is True
        shared_nan_arr = None

        for i, col in enumerate(columns_list):
            if not missing_mask[i]:
                arrays.append(data[col])
            else:
                if not fill_missing:
                    # Leave scalar NaN to let _homogenize handle dtype-specific casting
                    arrays.append(np.nan)
                else:
                    if dtype is not None:
                        arr = sanitize_array(np.nan, index, dtype=dtype)
                        arrays.append(arr)
                    else:
                        if shared_nan_arr is None:
                            nan_dtype = np.dtype("object")
                            shared_nan_arr = construct_1d_arraylike_from_scalar(
                                np.nan, len(index), nan_dtype
                            )
                        if copy:
                            arrays.append(shared_nan_arr)
                        else:
