        The assigned value has to be a list-like object. All items must be

        See Also
        --------
        rename_categories : Rename categories.
        reorder_categories : Reorder categories.
        add_categories : Add new categories.
        remove_categories : Remove the specified categories.
        remove_unused_categories : Remove categories which are not used.
        set_categories : Set the categories to the specified ones.

        Examples
        --------
        For :class:`pandas.Series`:

        >>> ser = pd.Series(["a", "b", "c", "a"], dtype="category")
        >>> ser.cat.categories
        Index(['a', 'b', 'c'], dtype='object')

        >>> raw_cat = pd.Categorical(["a", "b", "c", "A"], categories=["b", "c", "d"])
        >>> ser = pd.Series(raw_cat)
        >>> ser.cat.categories
        Index(['b', 'c', 'd'], dtype='object')

        For :class:`pandas.Categorical`:

        >>> cat = pd.Categorical(["a", "b"], ordered=True)

    @property
    def categories(self) -> Index:
        """
        The categories of this categorical.

        Setting assigns new values to each category (effectively a rename of
        each individual category).

        The assigned value has to be a list-like object. All items must be
        unique and the number of items in the new categories must be the same
        as the number of items in the old categories.

        Raises
        ------
        ValueError
            If the new categories do not validate as categories or if the
            number of new categories is unequal the number of old categories

        See Also
        --------
        rename_categories : Rename categories.
        reorder_categories : Reorder categories.
        add_categories : Add new categories.
        remove_categories : Remove the specified categories.
        remove_unused_categories : Remove categories which are not used.
        set_categories : Set the categories to the specified ones.

        Examples
        --------
        For :class:`pandas.Series`:

        >>> ser = pd.Series(["a", "b", "c", "a"], dtype="category")
        >>> ser.cat.categories
        Index(['a', 'b', 'c'], dtype='object')

        >>> raw_cat = pd.Categorical(["a", "b", "c", "A"], categories=["b", "c", "d"])
        >>> ser = pd.Series(raw_cat)
        >>> ser.cat.categories
        Index(['b', 'c', 'd'], dtype='object')

        For :class:`pandas.Categorical`:

        >>> cat = pd.Categorical(["a", "b"], ordered=True)
        >>> cat.categories
        Index(['a', 'b'], dtype='object')

        For :class:`pandas.CategoricalIndex`:

        >>> ci = pd.CategoricalIndex(["a", "c", "b", "a", "c", "b"])
        >>> ci.categories
        Index(['a', 'b', 'c'], dtype='object')

        >>> ci = pd.CategoricalIndex(["a", "c"], categories=["c", "b", "a"])
        >>> ci.categories
        Index(['c', 'b', 'a'], dtype='object')
        """
        from pandas import RangeIndex

        categories = self.dtype.categories
        if (
            isinstance(categories, RangeIndex)
            or len(categories) == 0
            or not categories.is_monotonic_increasing
            or not categories.is_unique
        ):
            return categories

        # try to preserve RangeIndex when possible for memory efficiency
        diff = categories[1] - categories[0]
        if (
            diff != 0
            and len(categories) > 1
            and categories.isna().sum() == 0
        ):
            try:
                # Check if this is a regular arithmetic progression
                expected = RangeIndex(
                    start=categories[0],
                    stop=categories[0] + len(categories) * diff,
                    step=diff,
                )
                if categories.equals(expected):
                    return expected
            except (TypeError, ValueError, OverflowError):
                pass
        return categories
