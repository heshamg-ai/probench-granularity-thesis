            # Fast path for float dtypes with na_value=np.nan: let pyarrow

            if (
                dtype is not None
                and np.issubdtype(np.dtype(dtype), np.floating)
                and lib.is_float(na_value)
                and np.isnan(na_value)

                result = np.asarray(self._pa_array, dtype=dtype)
                if copy:
                    result = result.copy()
                return result

            data[self.isna()] = na_value
            return np.asarray(data._pa_array, dtype=dtype)

            result = np.asarray(self._pa_array, dtype=dtype)
