def _get_empty_indexer() -> tuple[npt.NDArray[np.intp], npt.NDArray[np.intp]]:
    """Return empty join indexers."""    return (        np.array([], dtype=np.intp),        np.array([], dtype=np.intp),    )
def _get_join_indexers_monotonic(    left: Index, right: Index, how: JoinHow) -> tuple[npt.NDArray[np.intp], npt.NDArray[np.intp]]:
    """    Fast path for joining monotonic, sorted, no-NA keys using searchsorted.    Parameters    ----------    left : Index        Left join key, already validated as monotonic increasing with no NAs.    right : Index        Right join key, already validated as monotonic increasing with no NAs.    how : {'inner', 'outer', 'left', 'right'}        Type of join.    Returns    -------    np.ndarray[np.intp]        Left indexer.    np.ndarray[np.intp]        Right indexer.    """
    # Get the underlying join targets for comparison    lv = left._get_join_target()    rv = right._get_join_target()
    if how == "inner":        # Find intersection: values present in both        # Use searchsorted to find matching regions        left_pos = np.searchsorted(rv, lv, side="left")        right_pos = np.searchsorted(lv, rv, side="left")
        # For each left element, check if it exists in right        # It exists if left_pos points to a valid position and the value matches        left_in_right = (left_pos < len(rv)) & (rv[left_pos] == lv)
        # For each right element, check if it exists in left        right_in_left = (right_pos < len(lv)) & (lv[right_pos] == rv)

        # Build indexers for matches only        left_idx = np.nonzero(left_in_right)[0]        right_idx = left_pos[left_in_right]
        return left_idx.astype(np.intp), right_idx.astype(np.intp)

    elif how == "left":        # For each left element, find its position in right (if any)        left_pos = np.searchsorted(rv, lv, side="left")        left_in_right = (left_pos < len(rv)) & (rv[left_pos] == lv)
        left_idx = np.arange(len(lv), dtype=np.intp)        right_idx = np.where(left_in_right, left_pos, -1)        return left_idx, right_idx

    elif how == "right":        # Mirror of left: for each right element, find its position in left        right_pos = np.searchsorted(lv, rv, side="left")        right_in_left = (right_pos < len(lv)) & (lv[right_pos] == rv)
        right_idx = np.arange(len(rv), dtype=np.intp)        left_idx = np.where(right_in_left, right_pos, -1)        return left_idx, right_idx

    elif how == "outer":        # Merge two sorted arrays        # Use merge-sort-like approach        left_idx = []        right_idx = []        i = j = 0        while i < len(lv) and j < len(rv):            if lv[i] < rv[j]:                left_idx.append(i)                right_idx.append(-1)                i += 1            elif lv[i] > rv[j]:                left_idx.append(-1)                right_idx.append(j)                j += 1            else:  # equal                # Handle duplicates: need to emit all combinations                # Find run of equal values in left                i_start = i                while i < len(lv) and lv[i] == lv[i_start]:                    i += 1                # Find run of equal values in right                j_start = j                while j < len(rv) and rv[j] == rv[j_start]:                    j += 1                # Emit cartesian product of the runs                for ii in range(i_start, i):                    for jj in range(j_start, j):                        left_idx.append(ii)                        right_idx.append(jj)
        # Remaining elements        while i < len(lv):            left_idx.append(i)            right_idx.append(-1)            i += 1        while j < len(rv):            left_idx.append(-1)            right_idx.append(j)            j += 1
        return (            np.array(left_idx, dtype=np.intp),            np.array(right_idx, dtype=np.intp),        )
    else:        raise ValueError(f"how must be 'inner', 'outer', 'left', or 'right', got {how}")
