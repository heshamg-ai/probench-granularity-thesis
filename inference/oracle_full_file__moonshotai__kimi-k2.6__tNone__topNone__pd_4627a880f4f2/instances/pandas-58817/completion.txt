            return _unstack_frame(obj, level, fill_value=fill_value, sort=sort)
        else:

                from pandas.core.reshape.concat import concat

                obj_idx = obj.index
                obj_columns = obj.columns
                if obj_idx.nlevels != 1 or obj_columns.nlevels != 1:
                    # fallback to the default path
                    return obj.T.stack()

                # we can do a much faster path similar to stack but for unstack.
                # Unstack on a simple DataFrame is essentially:
                # Make each unique column value a new column, index becomes the

                from pandas import RangeIndex

                    pass
