def chi2(X, y):
    """Compute chi-squared stats between each non-negative feature and class.

    This score can be used to select the n_features features with the
    highest values for the test chi-squared statistic from X, which must
    contain only non-negative features such as booleans or frequencies
    (e.g., term counts in document classification), relative to the classes.

    Recall that the chi-square test measures dependence between stochastic
    variables, so using this function "weeds out" the features that are the
    most likely to be independent of class and therefore irrelevant for
    classification.

    Read more in the :ref:`User Guide <univariate_feature_selection>`.

    Parameters
    ----------
    X : {array-like, sparse matrix} of shape (n_samples, n_features)
        Sample vectors.

    y : array-like of shape (n_samples,)
        Target vector (class labels).

    Returns
    -------
    chi2 : ndarray of shape (n_features,)
        Chi2 statistics for each feature.

    p_values : ndarray of shape (n_features,)
        P-values for each feature.

    Notes
    -----
    Complexity of this algorithm is O(n_classes * n_features).

    See Also
    --------
    f_classif : ANOVA F-value between label/feature for classification tasks.
    f_regression : F-value between label/feature for regression tasks.
    """

    # XXX: we might want to do some of the following in logspace instead for
    # numerical stability.
    X = check_array(X, accept_sparse="csr")
    if np.any((X.data if issparse(X) else X) < 0):
        raise ValueError("Input X must be non-negative.")

    Y = LabelBinarizer().fit_transform(y)
    if Y.shape[1] == 1:
        Y = np.append(1 - Y, Y, axis=1)

    observed = safe_sparse_dot(Y.T, X)  # n_classes * n_features

    feature_count = X.sum(axis=0).reshape(1, -1)
    class_prob = Y.mean(axis=0).reshape(1, -1)
    expected = np.dot(class_prob.T, feature_count)

    return _chisquare(observed, expected)

def chi2(X, y):
    """Compute chi-squared stats between each non-negative feature and class.

    This score can be used to select the n_features features with the
    highest values for the test chi-squared statistic from X, which must
    contain only non-negative features such as booleans or frequencies
    (e.g., term counts in document classification), relative to the classes.

    Recall that the chi-square test measures dependence between stochastic
    variables, so using this function "weeds out" the features that are the
    most likely to be independent of class and therefore irrelevant for
    classification.

    Read more in the :ref:`User Guide <univariate_feature_selection>`.

    Parameters
    ----------
    X : {array-like, sparse matrix} of shape (n_samples, n_features)
        Sample vectors.

    y : array-like of shape (n_samples,)
        Target vector (class labels).

    Returns
    -------
    chi2 : ndarray of shape (n_features,)
        Chi2 statistics for each feature.

    p_values : ndarray of shape (n_features,)
        P-values for each feature.

    Notes
    -----
    Complexity of this algorithm is O(n_classes * n_features).

    See Also
    --------
    f_classif : ANOVA F-value between label/feature for classification tasks.
    f_regression : F-value between label/feature for regression tasks.
    """

    # XXX: we might want to do some of the following in logspace instead for
    # numerical stability.
    X = check_array(X, accept_sparse="csr")
    if np.any((X.data if issparse(X) else X) < 0):
        raise ValueError("Input X must be non-negative.")

    # Encode class labels to integers using factorize, which is more
    # memory-efficient than LabelBinarizer for many unique classes
    classes, y_encoded = np.unique(y, return_inverse=True)
    n_classes = len(classes)
    n_samples = X.shape[0]

    if n_classes == 1:
        # Only one class, chi2 is undefined; return zeros
        n_features = X.shape[1]
        return np.zeros(n_features, dtype=np.float64), np.ones(
            n_features, dtype=np.float64
        )

    # observed = class count matrix (n_classes, n_features)
    if issparse(X):
        from scipy.sparse import csr_matrix

        # For sparse matrices, use a more memory-efficient approach
        # Convert to CSR for fast row slicing if not already
        X = csr_matrix(X)
        # Sum features for each class using the encoded labels
        # We can use np.add.reduceat for efficient grouped summation
        # Sort by class index to use reduceat
        sort_order = np.argsort(y_encoded)
        y_sorted = y_encoded[sort_order]
        # Find boundaries where class changes
        class_boundaries = np.concatenate(
            ([0], np.where(np.diff(y_sorted) != 0)[0] + 1, [n_samples])
        )
        # Reorder X according to sorted classes
        X_sorted = X[sort_order]
        # Compute observed using reduceat for each feature
        observed = np.zeros((n_classes, X.shape[1]), dtype=np.float64)
        for i in range(n_classes):
            start = class_boundaries[i]
            end = class_boundaries[i + 1]
            observed[i] = np.asarray(X_sorted[start:end].sum(axis=0)).ravel()
    else:
        # Dense case: use bincount for efficient aggregation
        # For each feature, sum values per class
        # observed[c, j] = sum of X[i, j] for all i where y_encoded[i] == c
        # We can vectorize this using advanced indexing
        observed = np.zeros((n_classes, X.shape[1]), dtype=np.float64)
        for c in range(n_classes):
            mask = y_encoded == c
            if np.any(mask):
                observed[c] = X[mask].sum(axis=0)

    feature_count = np.asarray(X.sum(axis=0)).reshape(1, -1)
    # class_prob: fraction of samples in each class
    class_counts = np.bincount(y_encoded, minlength=n_classes).reshape(-1, 1)
    class_prob = class_counts / n_samples
    expected = np.dot(class_prob, feature_count)

    return _chisquare(observed, expected)
