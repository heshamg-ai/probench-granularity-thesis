    def _factorize_by_values(
        self,
        left_by_values: list[ArrayLike],
        right_by_values: list[ArrayLike],
    ) -> tuple[np.ndarray, np.ndarray]:
        """
        Factorize multiple by-values into integer codes for efficient matching.

        This is much faster than creating tuple arrays, especially for large
        datasets with multiple by-keys.

        Parameters
        ----------
        left_by_values : list[ArrayLike]
        right_by_values : list[ArrayLike]

        Returns
        -------
        tuple[np.ndarray, np.ndarray]
            Factorized integer arrays for left and right by-values.
        """
        # Fast path: factorize each key and combine using _get_join_keys
        # This avoids creating Python tuples and uses fast integer arithmetic
        llab: list[npt.NDArray[np.int64 | np.intp]] = []
        rlab: list[npt.NDArray[np.int64 | np.intp]] = []
        shape: list[int] = []

        for lbv, rbv in zip(left_by_values, right_by_values):
            # Factorize each pair of by-values
            lbv = extract_array(lbv, extract_numpy=True)
            rbv = extract_array(rbv, extract_numpy=True)

            # Handle ExtensionArrays by extracting underlying data when possible
            if isinstance(lbv, NDArrayBackedExtensionArray):
                lbv = lbv._ndarray
            if isinstance(rbv, NDArrayBackedExtensionArray):
                rbv = rbv._ndarray

            # Use pandas' efficient factorization
            lcodes, rcodes, count = _factorize_keys(lbv, rbv, sort=False)
            llab.append(lcodes.astype(np.intp, copy=False))
            rlab.append(rcodes.astype(np.intp, copy=False))
            shape.append(count)

        # Combine multiple keys into single integer keys using fast arithmetic
        # This is the same approach used in regular merge operations
        lkey, rkey = _get_join_keys(llab, rlab, tuple(shape), sort=False)

        # Final factorization to dense space
        lkey, rkey, _ = _factorize_keys(lkey, rkey, sort=False)

        # Ensure we return intp arrays for the Cython functions
        return (
            lkey.astype(np.intp, copy=False),
            rkey.astype(np.intp, copy=False),
        )
