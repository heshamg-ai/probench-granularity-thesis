        Combine two Series objects by filling null values in one Series with
        non-null values from the other Series. Result index will be the union
        of the two indexes.

        Parameters
        ----------
        other : Series
            The value(s) to be used for filling null values.

        Returns
        -------
        Series
            The result of combining the provided Series with the other object.

        See Also
        --------
        Series.combine : Perform element-wise operation on two Series
            using a given function.

        Examples
        --------
        >>> s1 = pd.Series([1, np.nan])
        >>> s2 = pd.Series([3, 4, 5])
        >>> s1.combine_first(s2)
        0    1.0
        1    4.0
        2    5.0
        dtype: float64

        Null values still persist if the location of that null value
        does not exist in `other`

        >>> s1 = pd.Series({'falcon': np.nan, 'eagle': 160.0})
        >>> s2 = pd.Series({'eagle': 200.0, 'duck': 30.0})
        >>> s1.combine_first(s2)
        duck       30.0
        eagle     160.0
        falcon      NaN
        dtype: float64
        """
        from pandas.core.reshape.concat import concat

        if self.index.equals(other.index):
            # Fast path for aligned indexes
            if self.dtype.kind == "M" and other.dtype.kind != "M":
                other = to_datetime(other)
            return self.where(notna(self), other).__finalize__(self, method="combine_first")

        # Try to use concat for partially overlapping / misaligned indexes
        # This is faster than reindex + where for many cases
        try:
            result = concat([self, other], copy=False).__finalize__(self, method="combine_first")
            # Group by index and take the first non-null value from each group
            if not result.index.is_unique:
                # Need to deduplicate: for each index, take first non-null
                result = result.groupby(level=0, sort=False).first()
            return result
        except (ValueError, TypeError):
            # Fallback to original implementation if concat fails
            pass

        new_index = self.index.union(other.index)
        this = self.reindex(new_index, copy=False)
        other = other.reindex(new_index, copy=False)
        if this.dtype.kind == "M" and other.dtype.kind != "M":
            other = to_datetime(other)

        return this.where(notna(this), other).__finalize__(self, method="combine_first")
