    def update_dtype(self, dtype: str_type | CategoricalDtype) -> CategoricalDtype:
        """
        Returns a CategoricalDtype with categories and ordered taken from dtype
        if specified, otherwise falling back to self if unspecified

        Parameters
        ----------
        dtype : CategoricalDtype

        Returns
        -------
        new_dtype : CategoricalDtype
        """
        if isinstance(dtype, str) and dtype == "category":
            # dtype='category' should not change anything
            return self
        elif not self.is_dtype(dtype):
            raise ValueError(
                f"a CategoricalDtype must be passed to perform an update, "
                f"got {dtype!r}"
            )
        else:
            # from here on, dtype is a CategoricalDtype
            dtype = cast(CategoricalDtype, dtype)

            # update categories/ordered unless they've been explicitly passed as None
            new_categories = (
                dtype.categories if dtype.categories is not None else self.categories
            )
            new_ordered = dtype.ordered if dtype.ordered is not None else self.ordered

            return CategoricalDtype(new_categories, new_ordered)

    def update_dtype(self, dtype: str_type | CategoricalDtype) -> CategoricalDtype:
        """
        Returns a CategoricalDtype with categories and ordered taken from dtype
        if specified, otherwise falling back to self if unspecified

        Parameters
        ----------
        dtype : CategoricalDtype

        Returns
        -------
        new_dtype : CategoricalDtype
        """
        if isinstance(dtype, str):
            if dtype == "category":
                # dtype='category' should not change anything
                return self
            raise ValueError(
                f"a CategoricalDtype must be passed to perform an update, "
                f"got {dtype!r}"
            )
        elif not self.is_dtype(dtype):
            raise ValueError(
                f"a CategoricalDtype must be passed to perform an update, "
                f"got {dtype!r}"
            )
        else:
            # from here on, dtype is a CategoricalDtype
            dtype = cast(CategoricalDtype, dtype)

            # update categories/ordered unless they've been explicitly passed as None
            if dtype.categories is None and dtype.ordered is None:
                # fastpath: nothing to update
                return self
            if dtype.categories is None:
                # only ordered may have changed
                if dtype.ordered == self.ordered:
                    return self
                return CategoricalDtype._from_fastpath(self.categories, dtype.ordered)
            if dtype.ordered is None:
                # only categories may have changed
                if dtype.categories is self.categories:
                    return self
                return CategoricalDtype._from_fastpath(dtype.categories, self.ordered)
            # both may have changed
            if dtype.categories is self.categories and dtype.ordered == self.ordered:
                return self
            return CategoricalDtype._from_fastpath(dtype.categories, dtype.ordered)
