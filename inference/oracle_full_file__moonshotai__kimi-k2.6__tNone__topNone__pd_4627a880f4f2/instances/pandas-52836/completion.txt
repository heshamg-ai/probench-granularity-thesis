        elif (

            # We have EAs with the same dtype. We can preserve that dtype in transpose.
            dtype = dtypes[0]
            arr_type = dtype.construct_array_type()

            # For masked EAs with 2D support, we can transpose much more efficiently
            # by transposing the underlying data and mask arrays directly.
            from pandas.core.arrays.masked import BaseMaskedArray

            if issubclass(arr_type, BaseMaskedArray):
                mgr = self._mgr
                # Single block with masked array
                if len(mgr.blocks) == 1:
                    arr = mgr.blocks[0].values
                    if arr.ndim == 2:
                        # 2D masked array - transpose directly
                        result = type(self)(
                            arr.T,
                            index=self.columns,
                            columns=self.index,
                            copy=False,
                        )
