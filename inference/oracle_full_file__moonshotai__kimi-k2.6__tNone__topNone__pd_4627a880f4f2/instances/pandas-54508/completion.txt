    def _fast_xs_extension(self, loc: int) -> SingleBlockManager:
        """
        Optimized fast_xs for extension dtype blocks.

        Parameters
        ----------
        loc : int

        Returns
        -------
        SingleBlockManager
        """
        # Collect values from each block for this row
        values = []
        for blk in self.blocks:
            for i in range(len(blk.mgr_locs)):
                values.append(blk.iget((i, loc)))

        # Concatenate all values into a single array
        if len(values) == 1:
            result = values[0]
        else:
            # Use the first array's _concat_same_type if available
            first = values[0]
            if hasattr(first, '_concat_same_type'):
                result = type(first)._concat_same_type(values)
            else:
                result = np.concatenate([np.asarray(v) for v in values])

        bp = BlockPlacement(slice(0, len(result)))
        block = new_block(result, placement=bp, ndim=1)
        return SingleBlockManager(block, self.axes[0])
