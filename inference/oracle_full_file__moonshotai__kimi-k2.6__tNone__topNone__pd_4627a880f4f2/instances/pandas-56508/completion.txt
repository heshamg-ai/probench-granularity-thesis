    def _hash_pandas_object(
        self, *, encoding: str, hash_key: str, categorize: bool
    ) -> npt.NDArray[np.uint64]:
        # Since masking variations of the same value (e.g., 1 and NA)
        # would produce the same hash, we need to incorporate the mask
        # into the hash. We do this by hashing the underlying data and
        # then XORing with a hash of the mask.
        from pandas.core.util.hashing import hash_array

        values, _ = self._values_for_factorize()

        result = hash_array(
            values, encoding=encoding, hash_key=hash_key, categorize=categorize
        )
        mask = self._mask
        result[mask] = hash_array(
            mask, encoding=encoding, hash_key=hash_key, categorize=categorize
        )[mask]
        # GH#55677 XOR with mask hash to distinguish masked and unmasked values
        return result

    def _hash_pandas_object(
        self, *, encoding: str, hash_key: str, categorize: bool
    ) -> npt.NDArray[np.uint64]:
        from pandas.core.util.hashing import hash_array, hash_object_array

        mask = self._mask
        hasna = mask.any()

        if hasna:
            # Start with zeros for NA values, to be filled in later
            result = np.zeros(len(self), dtype=np.uint64)
            # Only hash non-missing values
            valid = ~mask
            if valid.any():
                # Hash the valid underlying data
                valid_data = self._data[valid]
                valid_result = hash_array(
                    valid_data, encoding=encoding, hash_key=hash_key, categorize=categorize
                )
                result[valid] = valid_result
            # For masked positions, hash a sentinel that depends on position.
            # Using the mask itself as data avoids materializing an object array.
            mask_hash = hash_object_array(
                np.arange(len(self), dtype=object), hash_key=hash_key, encoding=encoding
            )
            result[mask] = mask_hash[mask]
        else:
            # Fast path: no missing values, hash underlying data directly
            result = hash_array(
                self._data, encoding=encoding, hash_key=hash_key, categorize=categorize
            )

        return result
