    def _str_get(self, i: int):
        # Use utf8_codeunit_at which is much faster than slice + if_else.
        # It returns null for out-of-bounds indices, matching pandas behavior.
        if i >= 0:
            return type(self)(pc.utf8_codeunit_at(self._pa_array, i))
        else:
            # For negative indices, convert to positive by computing length + i
            # then use utf8_codeunit_at
            pos = pc.add(pc.utf8_length(self._pa_array), i)
            return type(self)(pc.utf8_codeunit_at(self._pa_array, pos))
