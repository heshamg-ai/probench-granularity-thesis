            max_iter=self.transform_max_iter,

        )

        # Update inner stats
        self._update_inner_stats(X, code, batch_size, step)

        # Update dictionary
        _update_dict(

            X,
            code,
            self._A,
            self._B,
            verbose=self.verbose,
            random_state=random_state,
            positive=self.positive_dict,
        )

        # Compute batch cost using residuals to avoid extra matrix multiply
        residual = X - code @ dictionary
        batch_cost = (0.5 * (residual * residual).sum() + self.alpha * np.sum(np.abs(code))) / batch_size

        return batch_cost
