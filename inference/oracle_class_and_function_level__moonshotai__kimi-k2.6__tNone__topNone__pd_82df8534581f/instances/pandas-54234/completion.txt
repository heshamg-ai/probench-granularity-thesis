def group_idxmax(values, codes, ngroups):
    # values: numpy array (may have NaN)
    # codes: int array
    # ngroups: int
    
    # Initialize
    best_pos = np.full(ngroups, -1, dtype=np.intp)
    
    # Loop through all elements (in Python, but very tight)
    for i in range(len(values)):
        c = codes[i]
        if c < 0:
            continue
        if best_pos[c] < 0 or values[i] > values[best_pos[c]]:
            best_pos[c] = i
    
    return best_pos

                sort_key = -work_values
            else:
                work_values[is_nan] = np.inf
                sort_key = work_values
        else:
            is_nan = None
            if how == "idxmax":
                sort_key = -values
            else:
                sort_key = values

        # Stable sort by (codes, sort_key)
        # lexsort: last key is primary sort
        order = np.lexsort((sort_key, codes))
        sorted_codes = codes[order]

        # Filter out dropped groups (codes == -1)
        valid_mask = sorted_codes >= 0
        if not valid_mask.any():
            result_values = np.empty(ngroups, dtype=object)
            result_values.fill(np.nan)
            return self._obj_1d_constructor(
                result_values,
                index=self.grouper.result_index,
                dtype=ser.index.dtype,
            )

        filtered_codes = sorted_codes[valid_mask]
        unique_codes, first_in_filtered = np.unique(filtered_codes, return_index=True)
        keep_positions = np.where(valid_mask)[0]
        first_indices = keep_positions[first_in_filtered]
        orig_pos = order[first_indices]

        # Check for all-NaN groups
        if is_float:
            invalid = is_nan[orig_pos]
        else:
            invalid = np.zeros(len(orig_pos), dtype=bool)

        # Fall back to slow path for exact all-NaN error semantics
        if invalid.any():
            raise NotImplementedError

        # Build result
        result_values = np.empty(ngroups, dtype=object)
        result_values.fill(np.nan)

        valid_groups = ~invalid
        if valid_groups.any():
            result_values[unique_codes[valid_groups]] = ser.index[orig_pos[valid_groups]]

        return self._obj_1d_constructor(
            result_values, index=self.grouper.result_index, dtype=ser.index.dtype
        )

    def _fast_dataframe_idxmax_idxmin(
        self,
        how: Literal["idxmax", "idxmin"],
    ):
        """
        Fast vectorized idxmax/idxmin for DataFrameGroupBy with axis=0, skipna=True.
        Processes each column using the same fast logic as Series.
        """
        obj = self._obj_with_exclusions

        codes, _, _ = self.grouper.group_info
        result_index = self.grouper.result_index

        result_data = {}
        for col in obj.columns:
            ser = obj[col]
            values = ser._values

            # Check if we can use fast path for this column
            if not isinstance(values, np.ndarray) or values.dtype.kind not in "iuf":
                raise NotImplementedError

            is_float = values.dtype.kind == "f"

            if is_float:
                is_nan = np.isnan(values)
                work_values = values.copy()
                if how == "idxmax":
                    work_values[is_nan] = -np.inf
                    sort_key = -work_values
                else:
                    work_values[is_nan] = np.inf
                    sort_key = work_values
            else:
                is_nan = None
                if how == "idxmax":
                    sort_key = -values
                else:
                    sort_key = values

            order = np.lexsort((sort_key, codes))
            sorted_codes = codes[order]

            valid_mask = sorted_codes >= 0
            if not valid_mask.any():
                col_result = np.empty(len(result_index), dtype=object)
                col_result.fill(np.nan)
                result_data[col] = col_result
                continue

            filtered_codes = sorted_codes[valid_mask]
            unique_codes, first_in_filtered = np.unique(filtered_codes, return_index=True)
            keep_positions = np.where(valid_mask)[0]
            first_indices = keep_positions[first_in_filtered]
            orig_pos = order[first_indices]

            if is_float:
                invalid = is_nan[orig_pos]
            else:
                invalid = np.zeros(len(orig_pos), dtype=bool)

            # Fall back to slow path for exact all-NaN error semantics
            if invalid.any():
                raise NotImplementedError

            col_result = np.empty(len(result_index), dtype=object)
            col_result.fill(np.nan)

            valid_groups = ~invalid
            if valid_groups.any():
                col_result[unique_codes[valid_groups]] = ser.index[orig_pos[valid_groups]]

            result_data[col] = col_result

        return obj._constructor(result_data, index=result_index)
