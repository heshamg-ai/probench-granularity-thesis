def transpose_homogenous_masked_arrays(arrs, dtype=None, copy: bool = False):
    """
    Perform a "fast" transpose for a DataFrame with a single masked dtype.

    Given a list of 1D masked arrays (column data), produce a list of 1D
    masked arrays (row data) for the transpose. The transpose is performed
    by directly manipulating the underlying data and mask arrays rather than
    going through the generic numpy object-array machinery.

    Parameters
    ----------
    arrs : list of BaseMaskedArray
        Arrays representing each column of the original DataFrame.
    dtype : ExtensionDtype, optional
        The dtype of the masked arrays. If None, inferred from arrs[0].
    copy : bool, default False
        Whether to copy the data.

    Returns
    -------
    list of BaseMaskedArray
        Arrays representing each row of the transposed DataFrame.
    """
    import numpy as np

    from pandas.core.arrays.masked import BaseMaskedArray

    if not arrs:
        return []

    if dtype is None:
        dtype = arrs[0].dtype

    ncols = len(arrs)
    nrows = len(arrs[0])

    # Stack underlying data and masks
    data_blocks = np.empty((ncols, nrows), dtype=dtype.numpy_dtype)
    mask_blocks = np.empty((ncols, nrows), dtype=bool)

    for i, arr in enumerate(arrs):
        data_blocks[i, :] = arr._data
        mask_blocks[i, :] = arr._mask

    # Transpose
    data_t = data_blocks.T
    mask_t = mask_blocks.T

    if copy:
        data_t = data_t.copy()
        mask_t = mask_t.copy()

    result = []
    for i in range(nrows):
        arr = type(arrs[0])._simple_new(data_t[i], mask_t[i], dtype=dtype)
        result.append(arr)

    return result
