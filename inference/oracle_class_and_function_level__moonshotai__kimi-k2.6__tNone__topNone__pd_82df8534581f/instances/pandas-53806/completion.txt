    def size(self) -> Series:
        result: Series | DataFrame
        ngroups = self.ngroups
        if ngroups == 0:
            return Series(dtype="int64")
        if len(self._groupings) == 1 and self._is_monotonic:
            # Fast path for single grouping key with known sort order.
            # This avoids the cost of factorization when we already have
            # a sorted group structure and can compute sizes via diff.
            values = self._get_compressed_labels()[0]
            if len(values) == 0:
                return Series(dtype="int64")
            # values are group ids in sorted order; compute run lengths
            # by finding where the group id changes.
            boundaries = np.concatenate(([0], np.where(np.diff(values) != 0)[0] + 1, [len(values)]))
            sizes = np.diff(boundaries)
            result = Series(sizes, index=np.arange(len(sizes)))
            result.index = self.result_index
            return result
        if isinstance(self._groupings[0].grouping_vector, ops.BaseGrouper):
            result = self._cython_operation("size", min_count=0, needs_mask=True)
        else:
            result = self._cython_operation("size", min_count=0)
        return result
