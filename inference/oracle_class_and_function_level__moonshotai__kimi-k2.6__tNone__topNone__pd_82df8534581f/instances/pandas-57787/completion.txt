        The categories of this categorical.

        Setting assigns new values to each category (effectively a rename of

        The assigned value has to be a list-like object. All items must be

        Raises
        ------
        ValueError
            If the new categories do not validate as categories or if the
            number of new categories is unequal the number of old categories

        See Also
        --------
        rename_categories : Rename categories.
        reorder_categories : Reorder categories.
        add_categories : Add new categories.
        remove_categories : Remove the specified categories.
        remove_unused_categories : Remove categories which are not used.
        set_categories : Set the categories to the specified ones.

        Examples
        --------
        For :class:`pandas.Series`:

        >>> ser = pd.Series(["a", "b", "c", "a"], dtype="category")
        >>> ser.cat.categories
        Index(['a', 'b', 'c'], dtype='object')

        >>> raw_cat = pd.Categorical(["a", "b", "c", "a"], categories=["b", "c", "d"])
        >>> ser = pd.Series(raw_cat)
        >>> ser.cat.categories
        Index(['b', 'c', 'd'], dtype='object')

        For :class:`pandas.Categorical`:

        >>> cat = pd.Categorical(["a", "b"], ordered=True)

        from pandas import RangeIndex

        categories = self.dtype.categories
        if categories is None:
            return categories

        # Fast path: if categories already is a RangeIndex, return it directly
        if isinstance(categories, RangeIndex):
            return categories

        # Try to return a RangeIndex if the categories represent a range

            if diff != 0:
                expected = RangeIndex(categories[0], categories[-1] + diff, diff)
                if len(expected) == len(categories):
                    try:
                        if (categories == expected).all():
                            return expected
                    except (TypeError, ValueError):
                        pass

                pass
