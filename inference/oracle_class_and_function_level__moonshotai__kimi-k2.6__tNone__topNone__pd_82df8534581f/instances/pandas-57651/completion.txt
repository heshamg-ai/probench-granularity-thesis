    def join(
        self,
        other: Index,
        *,
        how: str = "left",
        level: Any | None = None,
        return_indexers: bool = False,
        sort: bool = False,
    ) -> Index | tuple[Index, npt.NDArray[np.intp] | None, npt.NDArray[np.intp] | None]:
        """
        Compute join_index and indexers to conform data structures to the new index.

        Parameters
        ----------
        other : Index
        how : {'left', 'right', 'inner', 'outer'}
        level : int or level name, default None
        return_indexers : bool, default False
        sort : bool, default False
            Sort the join keys lexicographically in the result Index. If False,
            the order of the join keys depends on the join type (how keyword).

        Returns
        -------
        join_index, (left_indexer, right_indexer)
        """
        other = ensure_index(other)

        if isinstance(other, RangeIndex) and level is None:
            # Fast path for RangeIndex joins - use set operations where possible
            if how == "left":
                join_index = self
                lidx = None
                ridx = other.get_indexer_for(join_index)
            elif how == "right":
                join_index = other
                lidx = self.get_indexer_for(join_index)
                ridx = None
            elif how == "inner":
                join_index = self.intersection(other)
                if len(join_index) == 0:
                    lidx = np.array([], dtype=np.intp)
                    ridx = np.array([], dtype=np.intp)
                else:
                    lidx = self.get_indexer_for(join_index)
                    ridx = other.get_indexer_for(join_index)
            else:  # how == "outer"
                join_index = self.union(other)
                lidx = self.get_indexer_for(join_index)
                ridx = other.get_indexer_for(join_index)

            if sort and not join_index.is_monotonic_increasing:
                join_index, indexer = join_index.sort_values(return_indexer=True)
                if lidx is not None:
                    lidx = lidx.take(indexer)
                if ridx is not None:
                    ridx = ridx.take(indexer)

            if return_indexers:
                return join_index, lidx, ridx
            else:
                return join_index

        # Fall back to Index.join for non-RangeIndex or level joins
        return super().join(
            other, how=how, level=level, return_indexers=return_indexers, sort=sort
        )
