        elif is_object_dtype(dtype) and self._hasna:
            result = np.empty(len(self), dtype=object)
            mask = ~self.isna()
            result[mask] = np.asarray(self[mask]._pa_array)
        elif pa.types.is_null(self._pa_array.type):
            result = np.asarray(self._pa_array, dtype=dtype)
            if not isna(na_value):
                result[:] = na_value
            return result
        elif self._hasna and (
            # fast path: numeric types with NaN-compatible na_value can use
            # pyarrow's zero-copy conversion and then fill nulls in-place
            (
                pa.types.is_floating(pa_type)
                and is_float_dtype(dtype)
                and lib.is_float(na_value)
                and isna(na_value)
            )
        ):
            result = np.asarray(self._pa_array, dtype=dtype)
            if copy:
                result = result.copy()
            null_mask = self._pa_array.is_null()
            if null_mask.true_count > 0:
                # use arrow's null bitmap directly for efficient filling
                null_bitmap = null_mask.combine_chunks()
                # Arrow boolean array stores validity as a bitmap
                # Invert to get null positions and fill with na_value
                result[~np.asarray(null_bitmap, dtype=bool)] = na_value
            return result
        elif self._hasna:
            data = self.copy()
            data[self.isna()] = na_value
            return np.asarray(data._pa_array, dtype=dtype)
        else:
            result = np.asarray(self._pa_array, dtype=dtype)

        if copy:
            result = result.copy()
        if self._hasna:
            result[self.isna()] = na_value
        return result
