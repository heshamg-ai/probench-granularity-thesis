def transpose_homogeneous_pyarrow(
    arrays: Sequence[ArrowExtensionArray],
) -> list[ArrowExtensionArray]:
    """Transpose pyarrow arrays in a list, but faster.

    Input should be a list of 1-dim pyarrow-backed arrays of equal length and
    all have the same pyarrow type. The caller is responsible for ensuring
    validity of input data.
    """
    import pyarrow as pa

    dtype = arrays[0].dtype
    arr_type = dtype.construct_array_type()

    # Extract chunks from all arrays and concatenate into a single array
    all_chunks = []
    for arr in arrays:
        pa_arr = arr._pa_array
        if hasattr(pa_arr, "chunks"):
            all_chunks.extend(pa_arr.chunks)
        else:
            all_chunks.append(pa_arr)

    # Concatenate all values into a single pyarrow array
    concatenated = pa.concat_arrays(all_chunks)

    # Number of columns (original arrays) and rows (elements per array)
    ncols = len(arrays)
    nrows = len(arrays[0])

    # Create transposed arrays by slicing at stride intervals
    transposed_arrays: list[ArrowExtensionArray] = []
    for i in range(nrows):
        # Slice starting at i, taking every ncols-th element
        sliced = concatenated.slice(i, ncols).tolist()
        # Build the new row using stride indexing
        row_values = [sliced[j] for j in range(0, ncols)]
        # Actually we need every ncols-th element from the concatenated array
        # Let me redo: we need indices i, i+nrows, i+2*nrows, ... no wait
        # Original layout: arr0[0], arr0[1], ..., arr0[nrows-1], arr1[0], arr1[1], ...
        # We want: for row i, we need arr0[i], arr1[i], arr2[i], ...
        # These are at indices: i, i+nrows, i+2*nrows, ..., i+(ncols-1)*nrows

        # Use pyarrow's take for efficient indexing
        indices = pa.array([i + j * nrows for j in range(ncols)])
        row = concatenated.take(indices)
        transposed_arr = arr_type(row, dtype=dtype)
        transposed_arrays.append(transposed_arr)

    return transposed_arrays
