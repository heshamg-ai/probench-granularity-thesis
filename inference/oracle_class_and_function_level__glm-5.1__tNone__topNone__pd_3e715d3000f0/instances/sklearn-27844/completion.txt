    def predict(self, X):
        """Predict classes for X.

        Parameters
        ----------
        X : array-like, shape (n_samples, n_features)
            The input samples.

        Returns
        -------
        y : ndarray, shape (n_samples,)
            The predicted classes.
        """
        # TODO: This could be done in parallel
        # Optimization for binary classification: avoid full predict_proba
        if self.n_trees_per_iteration_ == 1:
            raw_predictions = self._raw_predict(X).ravel()
            # For binary classification, positive class is predicted when
            # the sigmoid of the raw prediction is >= 0.5, i.e., raw >= 0.
            encoded_classes = (raw_predictions >= 0).astype(np.intp)
        else:
            encoded_classes = np.argmax(self.predict_proba(X), axis=1)
        return self.classes_[encoded_classes]
