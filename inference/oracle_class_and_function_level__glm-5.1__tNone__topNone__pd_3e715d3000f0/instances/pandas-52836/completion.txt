import numpy as np


def transpose_homogenous_masked_arrays(arrays, dtype):
    """
    Transpose a list of homogeneous masked arrays efficiently.

    Combines the underlying data and mask into 2D arrays, transposes
    them, then constructs new masked arrays from the transposed data.
    """
    n_rows = len(arrays[0])
    n_cols = len(arrays)

    # Stack data into a 2D ndarray and transpose
    stacked_data = np.empty((n_cols, n_rows), dtype=dtype.type)
    for i, arr in enumerate(arrays):
        stacked_data[i] = arr._data
    transposed_data = stacked_data.T

    # Stack masks into a 2D ndarray and transpose
    stacked_mask = np.empty((n_cols, n_rows), dtype=bool)
    for i, arr in enumerate(arrays):
        stacked_mask[i] = arr._mask
    transposed_mask = stacked_mask.T

    # Construct new arrays from the transposed data and mask
    arr_type = type(arrays[0])
    new_arrays = []
    for i in range(n_rows):
        new_arrays.append(
            arr_type._simple_new(
                transposed_data[i].copy(), transposed_mask[i].copy(), dtype=dtype
            )
        )

    return new_arrays
