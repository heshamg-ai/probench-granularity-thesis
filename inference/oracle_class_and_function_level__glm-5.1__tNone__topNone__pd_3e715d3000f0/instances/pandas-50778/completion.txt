            # in order to avoid repeating the same computations

            masks = [
                compare_or_regex_search(values, s[0], regex=regex, mask=mask)
                for s in pairs
            ]

            masks = [missing.mask_missing(values, s[0]) for s in pairs]

        masks = [extract_bool_array(x) for x in masks]

        if using_cow and inplace:
            # TODO(CoW): Optimize
            rb = [self.copy()]
        else:
            rb = [self if inplace else self.copy()]
        for i, (src, dest) in enumerate(pairs):
            convert = i == src_len  # only convert once at the end
            new_rb: list[Block] = []

            # GH-39338: _replace_coerce can split a block into
            # single-column blocks, so track the index so we know
            # where to index into the mask
            for blk_num, blk in enumerate(rb):
                if len(rb) == 1:
                    m = masks[i]
                else:
                    mib = masks[i]
                    assert not isinstance(mib, bool)
                    m = mib[blk_num : blk_num + 1]

                # error: Argument "mask" to "_replace_coerce" of "Block" has
                # incompatible type "Union[ExtensionArray, ndarray[Any, Any], bool]";
                # expected "ndarray[Any, dtype[bool_]]"
                result = blk._replace_coerce(
                    to_replace=src,
                    value=dest,
                    mask=m,  # type: ignore[arg-type]
                    inplace=inplace,
                    regex=regex,
                )
                if convert and blk.is_object and not all(x is None for x in dest_list):
                    # GH#44498 avoid unwanted cast-back
                    result = extend_blocks([b.convert(copy=True) for b in result])
                new_rb.extend(result)

    @final
    def replace_list(
        self,
        src_list: Iterable[Any],
        dest_list: Sequence[Any],
        inplace: bool = False,
        regex: bool = False,
        using_cow: bool = False,
    ) -> list[Block]:
        """
        See BlockManager.replace_list docstring.
        """
        values = self.values

        if isinstance(values, Categorical):
            # TODO: avoid special-casing
            # GH49404
            if using_cow and inplace:
                # TODO(CoW): Optimize
                blk = self.copy()
            else:
                blk = self if inplace else self.copy()
            values = cast(Categorical, blk.values)
            values._replace(to_replace=src_list, value=dest_list, inplace=True)
            return [blk]

        pairs = [
            (x, y) for x, y in zip(src_list, dest_list) if self._can_hold_element(x)
        ]
        if not len(pairs):
            # shortcut, nothing to replace
            return [self] if inplace else [self.copy()]

        # Optimization: for non-string, non-regex, non-None-dest bulk replacements,
        # perform all replacements in a single pass over the values array
        # using putmask_inplace. This avoids O(n*m) overhead from repeatedly
        # iterating over the array for each (src, dest) pair when m is large.
        if (
            not regex
            and not is_string_dtype(values.dtype)
            and not isinstance(values, Categorical)
            and all(dest is not None for _, dest in pairs)
        ):
            masks = [missing.mask_missing(values, s[0]) for s in pairs]

            # Check if any replacements will actually occur
            any_mask = False
            for m in masks:
                if m.any():
                    any_mask = True
                    break

            if not any_mask:
                if using_cow:
                    return [self.copy(deep=False)]
                return [self] if inplace else [self.copy()]

            if using_cow:
                if inplace:
                    blk = self.copy(deep=self.refs.has_reference())
                else:
                    blk = self.copy()
            else:
                blk = self if inplace else self.copy()

            for i, (src, dest) in enumerate(pairs):
                m = masks[i]
                if m.any():
                    if blk._can_hold_element(dest):
                        putmask_inplace(blk.values, m, dest)
                    else:
                        # Need to upcast, fall back to original logic
                        # Rebuild remaining pairs
                        remaining_pairs = [(s, d) for s, d in pairs[i:] if blk._can_hold_element(s)]
                        if remaining_pairs:
                            # Reapply already-processed pairs on the new block
                            # then continue with the remaining
                            # This is rare, so falling back to _replace_coerce is fine
                            rb = [blk]
                            src_len_remaining = len(remaining_pairs) - 1
                            # Rebuild masks for remaining pairs
                            remaining_masks = masks[i:i + len(remaining_pairs)]
                            for j, (s, d) in enumerate(remaining_pairs):
                                convert = j == src_len_remaining
                                new_rb: list[Block] = []
                                for blk_num, b in enumerate(rb):
                                    if len(rb) == 1:
                                        mm = remaining_masks[j]
                                    else:
                                        mib = remaining_masks[j]
                                        mm = mib[blk_num : blk_num + 1]
                                    result = b._replace_coerce(
                                        to_replace=s,
                                        value=d,
                                        mask=mm,  # type: ignore[arg-type]
                                        inplace=inplace,
                                        regex=regex,
                                    )
                                    if convert and b.is_object and not all(x is None for x in dest_list):
                                        result = extend_blocks([bb.convert(copy=True) for bb in result])
                                    new_rb.extend(result)
                                rb = new_rb
                            return rb
                        return [blk]

            if not (blk.is_object and any(x is None for _, x in pairs)):
                blocks = blk.convert(copy=False, using_cow=using_cow)
            else:
                blocks = [blk]
            return blocks

        src_len = len(pairs) - 1

        if is_string_dtype(values.dtype):
            # Calculate the mask once, prior to the call of comp
            # in order to avoid repeating the same computations
            mask = ~isna(values)
            masks = [
                compare_or_regex_search(values, s[0], regex=regex, mask=mask)
                for s in pairs
            ]
        else:
            # GH#38086 faster if we know we dont need to check for regex
            masks = [missing.mask_missing(values, s[0]) for s in pairs]

        masks = [extract_bool_array(x) for x in masks]

        if using_cow and inplace:
            # TODO(CoW): Optimize
            rb = [self.copy()]
        else:
            rb = [self if inplace else self.copy()]
        for i, (src, dest) in enumerate(pairs):
            convert = i == src_len  # only convert once at the end
            new_rb: list[Block] = []

            # GH-39338: _replace_coerce can split a block into
            # single-column blocks, so track the index so we know
            # where to index into the mask
            for blk_num, blk in enumerate(rb):
                if len(rb) == 1:
                    m = masks[i]
                else:
                    mib = masks[i]
                    assert not isinstance(mib, bool)
                    m = mib[blk_num : blk_num + 1]

                # error: Argument "mask" to "_replace_coerce" of "Block" has
                # incompatible type "Union[ExtensionArray, ndarray[Any, Any], bool]";
                # expected "ndarray[Any, dtype[bool_]]"
                result = blk._replace_coerce(
                    to_replace=src,
                    value=dest,
                    mask=m,  # type: ignore[arg-type]
                    inplace=inplace,
                    regex=regex,
                )
                if convert and blk.is_object and not all(x is None for x in dest_list):
                    # GH#44498 avoid unwanted cast-back
                    result = extend_blocks([b.convert(copy=True) for b in result])
                new_rb.extend(result)
