            # levels, avoiding expensive round-trips through _get_level_values
            # and Index.append which re-factorize each level independently.
            new_levels = []
            new_codes = []
            names = []

            for i in range(self.nlevels):
                lev = self.levels[i]
                level_codes = self.codes[i]
                lev_name = self._names[i]

                # Accumulate level, codes, and name from each 'other' MultiIndex
                others_lev = []
                others_codes = []
                others_name = []
                for o in other:
                    others_lev.append(o.levels[i])
                    others_codes.append(o.codes[i])
                    others_name.append(o._names[i])

                # Create a unified level by taking the union of all levels
                # This preserves self.levels ordering and appends new values
                # from others, exactly matching the factorization result from
                # _get_level_values -> Index.append
                combined_lev = lev
                for o_lev in others_lev:
                    combined_lev = combined_lev.union(o_lev, sort=False)

                # Remap self's codes into the combined level
                if combined_lev is lev:
                    # Fast path: no new values in others, combined_lev is the
                    # exact same object
                    combined_codes = [level_codes]
                else:
                    indexer = lev.get_indexer_for(combined_lev)
                    combined_codes = [indexer[level_codes]]

                # Remap each other's codes into the combined level
                for o_lev, o_codes in zip(others_lev, others_codes):
                    if o_lev is combined_lev:
                        # Fast path: exact same level object
                        combined_codes.append(o_codes)
                    elif o_lev.equals(combined_lev):
                        # Fast path: semantically equal levels
                        combined_codes.append(o_codes)
                    elif len(combined_lev) == len(lev) and combined_lev.equals(lev):
                        # combined_lev collapsed back to just lev
                        indexer = lev.get_indexer_for(combined_lev)
                        combined_codes.append(indexer[o_codes])
