    def _apply_fast_with_group_keys(self, f, *args, **kwargs):
        """
        Optimized path for DataFrame GroupBy.apply with many groups and group_keys=True.

        Avoids the overhead of concatting individual DataFrames with MultiIndex
        by fetching all group slices, applying the function, and building the
        final BlockManager and MultiIndex in one pass.
        """
        obj = self.obj
        group_indices = self.grouper.indices

        # Pre-allocate lists for the block manager concatenation
        mgrs_indexers = []
        keys = []

        # Construct the MultiIndex levels and codes simultaneously
        group_names = list(group_indices.keys())
        level_values = Index(group_names, name=self.grouper.names[0])
        level_codes = []
        original_index_codes = []

        # Get the columns / other axis once
        new_axes = list(obj.axes)
        columns = new_axes[1]

        for name in group_names:
            indices = group_indices[name]
            group_df = obj.iloc[indices]

            res = f(group_df, *args, **kwargs)
            if isinstance(res, ABCSeries):
                res = res.to_frame()
            elif not isinstance(res, ABCDataFrame):
                # Fall back to slow path if the function doesn't return a DataFrame
                return self._python_apply_general(f, *args, **kwargs)

            # Track the group key codes
            code = level_values.get_loc(name)
            level_codes.extend([code] * len(res))

            # Track the original index codes
            original_index_codes.extend(res.index)

            # Prepare the block manager for concatenation
            mgrs_indexers.append((res._mgr, {}))
            keys.append(name)

        # Build the MultiIndex
        original_index = Index(original_index_codes)
        if isinstance(obj.index, MultiIndex):
            # Combine the new level with the existing MultiIndex levels
            new_levels = [level_values] + list(obj.index.levels)
            new_codes = [np.array(level_codes, dtype=np.intp)] + [
                np.tile(lab, len(group_indices)) for lab in obj.index.codes
            ]
            new_names = [self.grouper.names[0]] + list(obj.index.names)
            final_index = MultiIndex(
                levels=new_levels, codes=new_codes, names=new_names, verify_integrity=False
            )
        else:
            final_index = MultiIndex.from_arrays(
                [level_values.take(level_codes), original_index],
                names=[self.grouper.names[0], obj.index.name],
            )

        new_axes[0] = final_index

        # Perform a fast concatenation of the BlockManagers
        from pandas.core.internals.concat import concatenate_managers

        new_data = concatenate_managers(
            mgrs_indexers, new_axes, concat_axis=0, copy=True
        )

        result = self.obj._constructor(new_data)
        return result.__finalize__(self, method="apply")
