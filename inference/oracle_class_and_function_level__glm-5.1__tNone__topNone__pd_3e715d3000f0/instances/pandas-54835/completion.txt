    @cache_readonly
    def is_monotonic_decreasing(self) -> bool:
        """
        Return a boolean if the values are equal or decreasing.
        """
        if len(self) == 0:
            return True

        # Check if all first-level codes are equal
        first_code = self.codes[0][0]
        if not (self.codes[0] == first_code).all():
            return self[::-1].is_monotonic_increasing

        # All first-level codes are equal, so the effective length for
        # the reverse check is determined by the first level's run
        if first_code == -1:
            return self[::-1].is_monotonic_increasing
        first_level = self.levels[0]
        first_value = first_code
        start_idx = first_level.get_loc(first_value)
        if isinstance(start_idx, int):
            end_idx = start_idx + 1
        elif isinstance(start_idx, slice):
            end_idx = start_idx.stop
        else:
            return self[::-1].is_monotonic_increasing

        effective_len = len(self) - (len(first_level) - end_idx)
        if effective_len <= 0:
            return True

        return self[:effective_len][::-1].is_monotonic_increasing
