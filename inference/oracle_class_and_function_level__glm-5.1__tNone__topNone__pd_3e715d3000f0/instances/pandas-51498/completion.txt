        """return block for the diff of the values"""

    def round(self, decimals: int) -> list[Block]:
        """
        Round numeric values to the given number of decimals.

        Non-numeric blocks are returned as-is (no-op).
        """
        if not (is_integer_dtype(self.dtype) or is_float_dtype(self.dtype)):
            return [self]
        values = self.values
        if isinstance(values, ExtensionArray):
            values = values._round(decimals)
        else:
            values = np.round(values, decimals)
        return [self.make_block(values=values)]
