def transpose_homogeneous_pyarrow(
    pyarrow_arrays: Sequence["ArrowExtensionArray"],
) -> list["ArrowExtensionArray"]:
    """Transpose pyarrow arrays in a list, but faster.

    Input should be a list of 1-dim ArrowExtensionArrays of equal length and all have
    the same dtype. The caller is responsible for ensuring validity of input data.
    """
    import pyarrow

    from pandas.core.arrays.arrow import ArrowExtensionArray

    n_rows = len(pyarrow_arrays[0])
    n_cols = len(pyarrow_arrays)
    dtyp = pyarrow_arrays[0].dtype
    pa_type = pyarrow_arrays[0]._pa_array.type

    # Concatenate all chunked arrays into a single contiguous chunked array
    combined = pyarrow.concat_arrays(
        [arr._pa_array.combine_chunks() for arr in pyarrow_arrays]
    )
    # Create a 2D (n_cols x n_rows) array and transpose to (n_rows x n_cols)
    transposed = combined.reshape(n_cols, n_rows).to_numpy().T

    # Build transposed ArrowExtensionArrays row by row
    transposed_arrays: list[ArrowExtensionArray] = []
    for i in range(n_rows):
        pa_arr = pyarrow.array(transposed[i], type=pa_type, from_pandas=True)
        transposed_arrays.append(ArrowExtensionArray(pa_arr))

    return transposed_arrays
