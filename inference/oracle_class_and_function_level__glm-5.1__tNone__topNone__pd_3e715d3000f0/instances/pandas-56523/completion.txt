def get_join_indexers(
    left_keys: list[ArrayLike],
    right_keys: list[ArrayLike],
    sort: bool = False,
    how: JoinHow = "inner",
) -> tuple[npt.NDArray[np.intp], npt.NDArray[np.intp]]:
    """

    Parameters
    ----------
    left_keys : list[ndarray, ExtensionArray, Index, Series]
    right_keys : list[ndarray, ExtensionArray, Index, Series]
    sort : bool, default False
    how : {'inner', 'outer', 'left', 'right'}, default 'inner'

    Returns
    -------
    np.ndarray[np.intp]
        Indexer into the left_keys.
    np.ndarray[np.intp]
        Indexer into the right_keys.
    """
    assert len(left_keys) == len(
        right_keys
    ), "left_keys and right_keys must be the same length"

    # fast-path for empty left/right
    left_n = len(left_keys[0])
    right_n = len(right_keys[0])
    if left_n == 0:
        if how in ["left", "inner"]:
            return _get_empty_indexer()
        elif not sort and how in ["right", "outer"]:
            return _get_no_sort_one_missing_indexer(right_n, True)
    elif right_n == 0:
        if how in ["right", "inner"]:
            return _get_empty_indexer()
        elif not sort and how in ["left", "outer"]:
            return _get_no_sort_one_missing_indexer(left_n, False)

    # get left & right join labels and num. of levels at each location
    mapped = (
        _factorize_keys(left_keys[n], right_keys[n], sort=sort)
        for n in range(len(left_keys))
    )
    zipped = zip(*mapped)
    llab, rlab, shape = (list(x) for x in zipped)

    # get flat i8 keys from label lists
    lkey, rkey = _get_join_keys(llab, rlab, tuple(shape), sort)

    # factorize keys to a dense i8 space
    # `count` is the num. of unique keys
    # set(lkey) | set(rkey) == range(count)

    lkey, rkey, count = _factorize_keys(lkey, rkey, sort=sort)
    # preserve left frame order if how == 'left' and sort == False
    kwargs = {}
    if how in ("inner", "left", "right"):
        kwargs["sort"] = sort
    join_func = {
        "inner": libjoin.inner_join,
        "left": libjoin.left_outer_join,
        "right": lambda x, y, count, **kwargs: libjoin.left_outer_join(
            y, x, count, **kwargs
        )[::-1],
        "outer": libjoin.full_outer_join,
    }[how]

    # error: Cannot call function of unknown type
    return join_func(lkey, rkey, count, **kwargs)  # type: ignore[operator]

def get_join_indexers(
    left_keys: list[ArrayLike],
    right_keys: list[ArrayLike],
    sort: bool = False,
    how: JoinHow = "inner",
) -> tuple[npt.NDArray[np.intp], npt.NDArray[np.intp]]:
    """

    Parameters
    ----------
    left_keys : list[ndarray, ExtensionArray, Index, Series]
    right_keys : list[ndarray, ExtensionArray, Index, Series]
    sort : bool, default False
    how : {'inner', 'outer', 'left', 'right'}, default 'inner'

    Returns
    -------
    np.ndarray[np.intp]
        Indexer into the left_keys.
    np.ndarray[np.intp]
        Indexer into the right_keys.
    """
    assert len(left_keys) == len(
        right_keys
    ), "left_keys and right_keys must be the same length"

    # fast-path for empty left/right
    left_n = len(left_keys[0])
    right_n = len(right_keys[0])
    if left_n == 0:
        if how in ["left", "inner"]:
            return _get_empty_indexer()
        elif not sort and how in ["right", "outer"]:
            return _get_no_sort_one_missing_indexer(right_n, True)
    elif right_n == 0:
        if how in ["right", "inner"]:
            return _get_empty_indexer()
        elif not sort and how in ["left", "outer"]:
            return _get_no_sort_one_missing_indexer(left_n, False)

    # fast-path for monotonic sorted single-key merges
    if len(left_keys) == 1 and sort:
        from pandas.core.arrays import BaseMaskedArray
        from pandas.core.arrays.string_ import StringArray
        from pandas.core.arrays.arrow import ArrowExtensionArray

        lk = left_keys[0]
        rk = right_keys[0]

        # Check if both keys are monotonic increasing numpy arrays
        # (or zero-copy convertible) and have compatible dtypes
        def _is_fast_key(arr: ArrayLike) -> bool:
            if isinstance(arr, np.ndarray):
                return arr.dtype.kind in "iufbMm"
            if isinstance(arr, BaseMaskedArray):
                return arr.dtype.kind in "iufbMm"
            if isinstance(arr, (StringArray, ArrowExtensionArray)):
                return True
            return False

        if _is_fast_key(lk) and _is_fast_key(rk):
            from pandas import Index
            from pandas.core.algorithms import is_monotonic

            left_idx = Index(lk)
            right_idx = Index(rk)

            if (
                left_idx.dtype == right_idx.dtype
                and left_idx._can_use_libjoin
                and right_idx._can_use_libjoin
                and left_idx.is_monotonic_increasing
                and right_idx.is_monotonic_increasing
            ):
                if left_idx.is_unique and right_idx.is_unique:
                    # Both unique and monotonic: use fast index join
                    if how == "left":
                        left_indexer = np.arange(left_n, dtype=np.intp)
                        right_indexer = left_idx._left_indexer_unique(right_idx)
                        return left_indexer, right_indexer
                    elif how == "right":
                        right_indexer = np.arange(right_n, dtype=np.intp)
                        left_indexer = right_idx._left_indexer_unique(left_idx)
                        return left_indexer, right_indexer
                    elif how == "inner":
                        joined, left_indexer, right_indexer = left_idx._inner_indexer(
                            right_idx
                        )
                        return left_indexer, right_indexer
                    elif how == "outer":
                        joined, left_indexer, right_indexer = left_idx._outer_indexer(
                            right_idx
                        )
                        return left_indexer, right_indexer
                elif left_idx.is_unique or right_idx.is_unique:
                    # At least one unique and both monotonic: use left/right joiner
                    if how == "left":
                        joined, left_indexer, right_indexer = left_idx._left_indexer(
                            right_idx
                        )
                        return left_indexer, right_indexer
                    elif how == "right":
                        joined, right_indexer, left_indexer = right_idx._left_indexer(
                            left_idx
                        )
                        return left_indexer, right_indexer
                    elif how == "inner":
                        joined, left_indexer, right_indexer = left_idx._inner_indexer(
                            right_idx
                        )
                        return left_indexer, right_indexer
                    elif how == "outer":
                        joined, left_indexer, right_indexer = left_idx._outer_indexer(
                            right_idx
                        )
                        return left_indexer, right_indexer

    # get left & right join labels and num. of levels at each location
    mapped = (
        _factorize_keys(left_keys[n], right_keys[n], sort=sort)
        for n in range(len(left_keys))
    )
    zipped = zip(*mapped)
    llab, rlab, shape = (list(x) for x in zipped)

    # get flat i8 keys from label lists
    lkey, rkey = _get_join_keys(llab, rlab, tuple(shape), sort)

    # factorize keys to a dense i8 space
    # `count` is the num. of unique keys
    # set(lkey) | set(rkey) == range(count)

    lkey, rkey, count = _factorize_keys(lkey, rkey, sort=sort)
    # preserve left frame order if how == 'left' and sort == False
    kwargs = {}
    if how in ("inner", "left", "right"):
        kwargs["sort"] = sort
    join_func = {
        "inner": libjoin.inner_join,
        "left": libjoin.left_outer_join,
        "right": lambda x, y, count, **kwargs: libjoin.left_outer_join(
            y, x, count, **kwargs
        )[::-1],
        "outer": libjoin.full_outer_join,
    }[how]

    # error: Cannot call function of unknown type
    return join_func(lkey, rkey, count, **kwargs)  # type: ignore[operator]
