    def _verify_integrity(self, codes: list | None = None, levels: list | None = None):
        """
        Parameters
        ----------
        codes : optional list
            Codes to check for validity. Defaults to current codes.
        levels : optional list
            Levels to check for validity. Defaults to current levels.

        Raises
        ------
        ValueError
            If length of levels and codes don't match, if the codes for any
            level would exceed level bounds, or there are any duplicate levels.

        Returns
        -------
        new codes where code value = -1 if it corresponds to a
        NaN level.
        """
        # NOTE: Currently does not check, among other things, that cached
        # nlevels matches nor that sortorder matches actually sortorder.
        codes = codes or self.codes
        levels = levels or self.levels

        if len(levels) != len(codes):
            raise ValueError(
                "Length of levels and codes must match. NOTE: "
                "this index is in an inconsistent state."
            )
        codes_length = len(codes[0])

        # Pre-compute sortorder check once
        sortorder = self.sortorder
        if sortorder is not None:
            sortorder_depth = _lexsort_depth(self.codes, self.nlevels)
            if sortorder > sortorder_depth:
                raise ValueError(
                    "Value for sortorder must be inferior or equal to actual "
                    f"lexsort_depth: sortorder {sortorder} "
                    f"with lexsort_depth {sortorder_depth}"
                )

        # Single pass: combine validation checks with NaN handling
        new_codes = []
        for i, (level, level_codes) in enumerate(zip(levels, codes)):
            if len(level_codes) != codes_length:
                raise ValueError(
                    f"Unequal code lengths: {[len(code_) for code_ in codes]}"
                )
            if len(level_codes):
                code_max = level_codes.max()
                if code_max >= len(level):
                    raise ValueError(
                        f"On level {i}, code max ({code_max}) >= length of "
                        f"level ({len(level)}). NOTE: this index is in an "
                        "inconsistent state"
                    )
                code_min = level_codes.min()
                if code_min < -1:
                    raise ValueError(f"On level {i}, code value ({code_min}) < -1")

            # Combine uniqueness check with NaN detection in single pass
            try:
                null_mask = isna(level)
            except TypeError:
                # Object-dtype levels without NA values
                null_mask = None

            if null_mask is not None and null_mask.any():
                # NaN present - need to validate codes and update -1 codes
                if not level.is_unique:
                    raise ValueError(
                        f"Level values must be unique: {list(level)} on level {i}"
                    )
                code = np.where(null_mask[level_codes], -1, level_codes)
            else:
                # No NaN values - only need uniqueness check
                if not level.is_unique:
                    raise ValueError(
                        f"Level values must be unique: {list(level)} on level {i}"
                    )
                code = level_codes
                if isinstance(code, np.ndarray) and not code.flags.writeable:
                    code = code.copy()

            new_codes.append(code)

        return FrozenList(new_codes)
