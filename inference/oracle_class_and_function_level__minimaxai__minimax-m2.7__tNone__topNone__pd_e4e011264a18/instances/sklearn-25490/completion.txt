        # Compute code for this batch

            dictionary,
            algorithm=self._fit_algorithm,
            alpha=self.alpha,
            n_jobs=self.n_jobs,
            check_input=False,
            positive=self.positive_code,
            max_iter=self.transform_max_iter,
            verbose=self.verbose,

        # Compute batch cost efficiently using optimized operations
        # reconstruction = X - code @ dictionary
        # Use einsum for efficient computation avoiding large temporary arrays
        reconstruction = code @ dictionary
        reconstruction *= -1
        reconstruction += X
        # Compute ||X - CD||_F^2 efficiently

    def _update_inner_stats(self, X, code, batch_size, step):
        """Update the inner stats inplace."""
        if step < batch_size - 1:
            theta = (step + 1) * batch_size
        else:
            theta = batch_size**2 + step + 1 - batch_size
        beta = (theta + 1 - batch_size) / (theta + 1)

        self._A *= beta
        self._A += code.T @ code / batch_size
        self._B *= beta
        self._B += X.T @ code / batch_size

    def _update_inner_stats(self, X, code, batch_size, step):
        """Update the inner stats inplace."""
        if step < batch_size - 1:
            theta = (step + 1) * batch_size
        else:
            theta = batch_size**2 + step + 1 - batch_size
        beta = (theta + 1 - batch_size) / (theta + 1)

        # For small batch sizes, use more numerically stable incremental updates
        # to avoid large intermediate matrices from code.T @ code
        if batch_size <= 32:
            # For small batches: compute outer products incrementally
            # This is more cache-friendly and avoids large matrix mult
            self._A *= beta
            self._B *= beta
            # Use optimized outer product loop
            for i in range(batch_size):
                self._A += np.outer(code[i], code[i]) / batch_size
                self._B += np.outer(X[i], code[i]) / batch_size
        else:
            self._A *= beta
            self._A += code.T @ code / batch_size
            self._B *= beta
            self._B += X.T @ code / batch_size
