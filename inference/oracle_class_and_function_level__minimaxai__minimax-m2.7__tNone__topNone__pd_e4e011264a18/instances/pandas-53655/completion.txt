    def _str_get_dummies(self, sep: str = "|"):
        split = pc.split_pattern(self._pa_array, sep).combine_chunks()

        # Get unique values
        uniques = split.flatten().unique()
        uniques_sorted = uniques.take(pa.compute.array_sort_indices(uniques))
        n_unique = len(uniques_sorted)

        if n_unique == 0:
            result = type(self)(pa.array([], type=pa.bool_()))
            return result, uniques_sorted.to_pylist()

        n_rows = len(self)
        result_matrix = np.zeros((n_rows, n_unique), dtype=np.bool_)

        # Iterate over unique values - only call to_pylist() on smaller index arrays
        for i in range(n_unique):
            val = uniques_sorted[i]
            indices = pc.index_in_array(split, val)
            row_indices = indices.to_pylist()
            for row_idx in row_indices:
                result_matrix[row_idx, i] = True

        result = type(self)(pa.array(result_matrix))
        return result, uniques_sorted.to_pylist()
