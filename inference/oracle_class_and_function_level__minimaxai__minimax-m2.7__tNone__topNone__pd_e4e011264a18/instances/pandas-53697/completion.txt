        if not isinstance(other, (list, tuple)):
            other = [other]

        if all(
            (isinstance(o, MultiIndex) and o.nlevels == self.nlevels) for o in other
        ):
            # Fast path: if all levels are identical objects across all
            # MultiIndexes, we can directly concatenate codes without
            # materializing values
            all_levels_match = True
            for i in range(self.nlevels):
                self_level = self._levels[i]
                for o in other:
                    if o._levels[i] is not self_level:
                        all_levels_match = False
                        break
                if not all_levels_match:
                    break

            if all_levels_match:
                # Concatenate codes directly - all levels are identical
                codes = [level_codes for level_codes in self._codes]
                for o in other:
                    for i in range(self.nlevels):
                        codes[i] = np.concatenate([codes[i], o._codes[i]])
                return MultiIndex(
                    levels=self._levels,
                    codes=codes,
                    names=self._names,
                    sortorder=None,
                    verify_integrity=False,
                )

            # Fallback: use the original approach with _get_level_values
