def _round_field(values, name, freq):
    """Apply rounding operation to datetime-like values.

    Parameters
    ----------
    values : np.ndarray or dask.array-like
        Array-like container of datetime-like values
    name : str
        Name of datetime field to access (floor, ceil, round)
    freq : str
        Frequency string for rounding operations

    Returns
    -------
    rounded_values : same type as values
        Array-like of rounded datetime values

    """
    if is_duck_dask_array(values):
        from dask.array import map_blocks

        return map_blocks(_round_field, values, name, freq, dtype=values.dtype)
    elif not is_np_datetime_like(values.dtype):
        # Use CFTimeIndex directly for cftime to avoid Series overhead
        from xarray.coding.cftimeindex import CFTimeIndex

        values_as_cftimeindex = CFTimeIndex(values.ravel())
        result = getattr(values_as_cftimeindex, name)(freq)
        return result.values.reshape(values.shape)
    else:
        import pandas as pd

        return (
            pd.Series(values.ravel())
            .dt
            .__getattribute__(name)(freq)
            .values.reshape(values.shape)
        )
