    def predict(self, X):
        """Predict classes for X.

        Parameters
        ----------
        X : array-like, shape (n_samples, n_features)
            The input samples.

        Returns
        -------
        y : ndarray, shape (n_samples,)
            The predicted classes.
        """
        # For binary classification (n_trees_per_iteration_ == 1), raw_predictions
        # has shape (n_samples, 1). Since the loss is log_loss, the predicted class
        # depends on the sign of the raw prediction: if raw >= 0, the probability
        # of class 1 is >= 0.5, otherwise class 0. We can directly use this sign
        # comparison instead of computing sigmoid probabilities and taking argmax.
        raw_predictions = self._raw_predict(X)
        if self.n_trees_per_iteration_ == 1:
            encoded_classes = (raw_predictions.ravel() >= 0).astype(int)
        else:
            encoded_classes = np.argmax(
                self._loss.predict_proba(raw_predictions), axis=1
            )
        return self.classes_[encoded_classes]
