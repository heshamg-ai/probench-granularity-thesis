                if isinstance(result.dtype, ArrowDtype):
                    import pyarrow as pa

                    from pandas.compat import pa_version_under11p0

                    from pandas.core.arrays.arrow.array import ArrowExtensionArray

                    # Combine chunks for efficient processing
                    combined = result._pa_array.combine_chunks()
                    value_lengths = combined.value_lengths()
                    max_len = pa.compute.max(value_lengths).as_py()
                    min_len = pa.compute.min(value_lengths).as_py()

                    if result._hasna or min_len < max_len:
                        if min_len < max_len and not pa_version_under11p0:
                            # Use efficient list_slice to pad shorter lists
                            # This returns fixed-size lists which is more efficient
                            combined = pa.compute.list_slice(
                                combined,
                                start=0,
                                stop=max_len,
                                return_fixed_size_list=True,
                            )
                        elif result._hasna:
                            # ArrowExtensionArray.fillna doesn't work for list scalars
                            combined = combined.fill_null([None] * max_len)
                        if min_len < max_len and pa_version_under11p0:
                            # Fallback for older PyArrow versions
                            all_null = np.full(max_len, fill_value=None, dtype=object)
                            values = result.to_numpy()
                            new_values = []
                            for row in values:
                                if len(row) < max_len:
                                    nulls = all_null[: max_len - len(row)]
                                    row = np.append(row, nulls)
                                new_values.append(row)
                            pa_type = result._pa_array.type
                            combined = pa.array(new_values, type=pa_type)

                    if name is not None:
                        labels = name
                    else:
                        labels = range(max_len)

                    # Fast path: if lists have uniform length, extract columns directly
                    # using PyArrow compute functions instead of Python iteration
                    if min_len == max_len:
                        # Extract each column using list_element
                        columns_data = {}
                        combined_chunked = pa.chunked_array([combined])
                        for col_idx in range(max_len):
