        src_len = len(pairs) - 1

        # Optimization for non-string dtypes with many replacement pairs
        # Uses vectorized operations instead of per-pair mask creation
        if (
            not is_string_dtype(values.dtype)
            and len(pairs) > 10
            and not isinstance(values, ExtensionArray)
        ):
            # Create lookup mapping
            src_to_dest = {p[0]: p[1] for p in pairs}
            unique_src = list(src_to_dest.keys())
            src_arr = np.asarray(unique_src, dtype=object)

            # Find values that match any source value
            mask = np.isin(values, src_arr)
            if not mask.any():
                return [self] if inplace else [self.copy()]

            # Flatten for vectorized operations
            values_flat = values.ravel()
            mask_flat = mask.ravel()

            # Get unique values that need replacement
            values_to_replace = values_flat[mask_flat]
            unique_vals, inverse = np.unique(values_to_replace, return_inverse=True)

            # Check which unique values are in source
            in_src = np.isin(unique_vals, src_arr)
            if not in_src.any():
                return [self] if inplace else [self.copy()]

            # Build replacement array
            replacements = np.array(
                [src_to_dest.get(v, v) for v in unique_vals], dtype=object
            )

            # Apply replacements
            if inplace:
                result_flat = values_flat
            else:
                result_flat = values_flat.copy()
            result_flat[mask_flat] = replacements[inverse]

            if inplace:
                return [self]
            else:
                result = result_flat.reshape(values.shape)
                return [self.make_block(result, refs=self.refs)]
