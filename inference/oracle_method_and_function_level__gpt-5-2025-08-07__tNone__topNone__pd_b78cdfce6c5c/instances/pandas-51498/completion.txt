    def round(self, decimals: int):
        arrays = self.arrays
        changed = False
        new_arrays: list[np.ndarray | ExtensionArray] = []
        for arr in arrays:
            new_arr = arr
            if isinstance(arr, np.ndarray):
                if np.issubdtype(arr.dtype, np.floating):
                    new_arr = np.round(arr, decimals)
                elif np.issubdtype(arr.dtype, np.integer):
                    if decimals < 0:
                        new_arr = np.round(arr, decimals).astype(arr.dtype, copy=False)
                    else:
                        new_arr = arr
                else:
                    new_arr = arr
            else:
                try:
                    new_arr = arr.round(decimals)  # type: ignore[attr-defined]
                except Exception:
                    from pandas import Series
                    try:
                        new_arr = Series(arr).round(decimals).array
                    except Exception:
                        new_arr = arr
            if new_arr is not arr:
                changed = True
            new_arrays.append(new_arr)
        if not changed:
            return self
        return type(self)(new_arrays, self._axes, verify_integrity=False)
