        # Precompute segment lengths and total length for efficient filling
        seg_lengths = [len(idx) for idx in indexes]
        total_len = sum(seg_lengths)

        for hlevel, level in zip(zipped, levels):
            # Vectorized mapping from keys to level codes, treating NaN as equal.
            mapped = level.get_indexer(hlevel)

            if (mapped == -1).any():
                lvl_isna = isna(level)
                if lvl_isna.any():
                    nan_pos = np.nonzero(lvl_isna)[0][0]
                    h_isna = isna(hlevel)
                    fill_mask = (mapped == -1) & h_isna
                    if fill_mask.any():
                        mapped[fill_mask] = nan_pos

                if (mapped == -1).any():
                    first_bad = int(np.nonzero(mapped == -1)[0][0])
                    bad_key = hlevel[first_bad]
                    raise ValueError(f"Key {bad_key} not in level {level}")

            # Build codes array by filling segments directly to avoid many small arrays
            codes = np.empty(total_len, dtype=mapped.dtype)
            pos = 0
            for code, seg_len in zip(mapped, seg_lengths):
                codes[pos:pos + seg_len] = code
                pos += seg_len
            codes_list.append(codes)
