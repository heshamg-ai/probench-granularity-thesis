        # Precompute mappings for faster per-variable lookup
        window_dim_map = {d: window_dims[i] for i, d in enumerate(self.dim)}
        stride_map = {d: strides[i] for i, d in enumerate(self.dim)}

        dataset = {}
        for key, da in self.obj.data_vars.items():
            # keeps rollings only for the dataset depending on self.dim
            dims = [d for d in self.dim if d in da.dims]
            if dims:
                wi = {d: window_dim_map[d] for d in dims}
                st = {d: stride_map[d] for d in dims}

                dataset[key] = self.rollings[key].construct(
                    window_dim=wi,
                    fill_value=fill_value,
                    stride=st,
                    keep_attrs=keep_attrs,
                )
            else:
                dataset[key] = da.copy()

            # as the DataArrays can be copied we need to delete the attrs
            if not keep_attrs:
                dataset[key].attrs = {}

        attrs = self.obj.attrs if keep_attrs else {}

        # Avoid unnecessary isel when all strides are 1
        if all(s == 1 for s in stride_map.values()):
            return Dataset(dataset, coords=self.obj.coords, attrs=attrs)

        indexers = {d: slice(None, None, s) for d, s in stride_map.items()}
        return Dataset(dataset, coords=self.obj.coords, attrs=attrs).isel(indexers)
