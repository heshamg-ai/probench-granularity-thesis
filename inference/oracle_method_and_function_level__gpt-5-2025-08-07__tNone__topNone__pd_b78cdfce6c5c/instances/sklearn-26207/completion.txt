    """Rescale data sample-wise by square root of sample_weight.

    For many linear models, this enables easy support for sample_weight because

        (y - X w)' S (y - X w)

    with S = diag(sample_weight) becomes

        ||y_rescaled - X_rescaled w||_2^2

    when setting

        y_rescaled = sqrt(S) y
        X_rescaled = sqrt(S) X

    Returns
    -------
    X_rescaled : {array-like, sparse matrix}

    y_rescaled : {array-like, sparse matrix}
    """

def _rescale_data(X, y, sample_weight):
    """Rescale data sample-wise by square root of sample_weight.

    For many linear models, this enables easy support for sample_weight because

        (y - X w)' S (y - X w)

    with S = diag(sample_weight) becomes

        ||y_rescaled - X_rescaled w||_2^2

    when setting

        y_rescaled = sqrt(S) y
        X_rescaled = sqrt(S) X

    Returns
    -------
    X_rescaled : {array-like, sparse matrix}

    y_rescaled : {array-like, sparse matrix}
    """
    n_samples = X.shape[0]
    sample_weight = np.asarray(sample_weight)
    if sample_weight.ndim == 0:
        sample_weight = np.full(n_samples, sample_weight, dtype=sample_weight.dtype)
    sample_weight_sqrt = np.sqrt(sample_weight)

    # Scale X
    if sparse.issparse(X):
        # Create a scaled copy to preserve original input when needed
        if sparse.isspmatrix_csr(X):
            X = X.copy()
            # Repeat weights for each non-zero entry per row
            X.data *= np.repeat(sample_weight_sqrt, np.diff(X.indptr))
        elif sparse.isspmatrix_csc(X):
            X = X.copy()
            # Scale by row weights using row indices
            X.data *= sample_weight_sqrt[X.indices]
        else:
            # Fallback for other formats
            sw_matrix = sparse.dia_matrix((sample_weight_sqrt, 0), shape=(n_samples, n_samples))
            X = safe_sparse_dot(sw_matrix, X)
    else:
        # Allocate a new array to mirror previous behavior (no in-place requirement)
        X = X * sample_weight_sqrt[:, np.newaxis]

    # Scale y
    if sparse.issparse(y):
        if sparse.isspmatrix_csr(y):
            y = y.copy()
            y.data *= np.repeat(sample_weight_sqrt, np.diff(y.indptr))
        elif sparse.isspmatrix_csc(y):
            y = y.copy()
            y.data *= sample_weight_sqrt[y.indices]
        else:
            sw_matrix = sparse.dia_matrix((sample_weight_sqrt, 0), shape=(n_samples, n_samples))
            y = safe_sparse_dot(sw_matrix, y)
    else:
        if y.ndim == 2:
            y = y * sample_weight_sqrt[:, np.newaxis]
        else:
            y = y * sample_weight_sqrt

    return X, y, sample_weight_sqrt
