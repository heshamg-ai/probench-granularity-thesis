        def blk_func(values: ArrayLike) -> ArrayLike:
            mask = isna(values)
            has_dropped_na = self._grouper.has_dropped_na

            if values.ndim == 1:
                # Fast-path: no NAs anywhere and no dropped NA groups
                if not has_dropped_na and not mask.any():
                    return values.copy()

                indexer = np.empty(values.shape, dtype=np.intp)
                col_func(out=indexer, mask=mask)
                return algorithms.take_nd(values, indexer)

            else:
                # We broadcast algorithms.take_nd analogous to
                #  np.take_along_axis
                if isinstance(values, np.ndarray):
                    # Fast-path: no NAs anywhere and no dropped NA groups
                    if not has_dropped_na and not mask.any():
                        return values.copy()

                    dtype = values.dtype
                    if has_dropped_na:
                        # dropped null groups give rise to nan in the result
                        dtype = ensure_dtype_can_hold_na(values.dtype)
                    out = np.empty(values.shape, dtype=dtype)
                else:
                    # Note: we only get here with backfill/pad,
                    #  so if we have a dtype that cannot hold NAs,
                    #  then there will be no -1s in indexer, so we can use
                    #  the original dtype (no need to ensure_dtype_can_hold_na)
                    out = type(values)._empty(values.shape, dtype=values.dtype)

                # Pre-allocate and reuse the indexer per row to reduce allocations
                row_indexer = np.empty(values.shape[1], dtype=np.intp)
                # Precompute which rows actually contain any missing values
                row_has_na = mask.any(axis=1)

                for i, value_element in enumerate(values):
                    # call group_fillna_indexer column-wise only when needed
                    if not has_dropped_na and not row_has_na[i]:
                        # No NA to fill and no dropped-na groups: just copy the row
                        out[i, :] = value_element
                    else:
                        col_func(out=row_indexer, mask=mask[i])
                        out[i, :] = algorithms.take_nd(value_element, row_indexer)
                return out
