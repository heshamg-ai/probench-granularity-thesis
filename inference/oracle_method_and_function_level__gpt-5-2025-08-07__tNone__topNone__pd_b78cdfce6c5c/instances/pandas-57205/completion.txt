    if columns is not None:
        # Fastpath: align dict values to provided columns without constructing
        # an intermediate Series when there are no missing keys and all values
        # are array-like (non-scalar). Fallback to the original path otherwise.
        use_fastpath = True
        try:
            for c in columns:
                if c not in data:
                    use_fastpath = False
                    break
        except TypeError:
            # Unhashable labels, cannot check membership reliably
            use_fastpath = False
