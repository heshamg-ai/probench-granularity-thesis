def _round_through_cftimeindex(values, name, freq):
    """Round CFTime-like arrays by constructing a CFTimeIndex and
    calling the corresponding rounding method (floor/ceil/round).
    Works with both NumPy and Dask arrays.
    """
    from xarray.coding.cftimeindex import CFTimeIndex

    if is_duck_dask_array(values):
        from dask.array import map_blocks

        def _block_round(block, name, freq):
            idx = CFTimeIndex(block.ravel())
            method = getattr(idx, name)
            out = method(freq).values
            return out.reshape(block.shape)

        return map_blocks(_block_round, values, name, freq, dtype=values.dtype)
    else:
        idx = CFTimeIndex(values.ravel())
        method = getattr(idx, name)
        out = method(freq).values
        return out.reshape(values.shape)
