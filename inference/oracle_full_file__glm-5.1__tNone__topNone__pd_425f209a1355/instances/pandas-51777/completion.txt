    def combine_first(self, other) -> Series:
        """
        Update null elements with value in the same location in 'other'.

        Combine two Series objects by filling null values in one Series with
        non-null values from the other Series. Result index will be the union
        of the two indexes.

        Parameters
        ----------
        other : Series
            The value(s) to be used for filling null values.

        Returns
        -------
        Series
            The result of combining the provided Series with the other object.

        See Also
        --------
        Series.combine : Perform element-wise operation on two Series
            using a given function.

        Examples
        --------
        >>> s1 = pd.Series([1, np.nan])
        >>> s2 = pd.Series([3, 4, 5])
        >>> s1.combine_first(s2)
        0    1.0
        1    4.0
        2    5.0
        dtype: float64

        Null values still persist if the location of that null value
        does not exist in `other`

        >>> s1 = pd.Series({'falcon': np.nan, 'eagle': 160.0})
        >>> s2 = pd.Series({'eagle': 200.0, 'duck': 30.0})
        >>> s1.combine_first(s2)
        duck       30.0
        eagle     160.0
        falcon      NaN
        dtype: float64
        """
        new_index = self.index.union(other.index)
        this = self.reindex(new_index, copy=False)
        other = other.reindex(new_index, copy=False)
        if this.dtype.kind == "M" and other.dtype.kind != "M":
            other = to_datetime(other)

        return this.where(notna(this), other)

    def combine_first(self, other) -> Series:
        """
        Update null elements with value in the same location in 'other'.

        Combine two Series objects by filling null values in one Series with
        non-null values from the other Series. Result index will be the union
        of the two indexes.

        Parameters
        ----------
        other : Series
            The value(s) to be used for filling null values.

        Returns
        -------
        Series
            The result of combining the provided Series with the other object.

        See Also
        --------
        Series.combine : Perform element-wise operation on two Series
            using a given function.

        Examples
        --------
        >>> s1 = pd.Series([1, np.nan])
        >>> s2 = pd.Series([3, 4, 5])
        >>> s1.combine_first(s2)
        0    1.0
        1    4.0
        2    5.0
        dtype: float64

        Null values still persist if the location of that null value
        does not exist in `other`

        >>> s1 = pd.Series({'falcon': np.nan, 'eagle': 160.0})
        >>> s2 = pd.Series({'eagle': 200.0, 'duck': 30.0})
        >>> s1.combine_first(s2)
        duck       30.0
        eagle     160.0
        falcon      NaN
        dtype: float64
        """
        if not isinstance(other, Series):
            other = self._constructor(other)

        new_index = self.index.union(other.index)

        # Fast path: identical indexes (including both empty)
        if new_index is self.index and self.index is other.index:
            # When indexes are identical objects, reindex is a no-op,
            # so we can use the original values directly without creating
            # reindexed copies.
            this_values = self._values
            other_values = other._values
        else:
            this = self.reindex(new_index, copy=False)
            other = other.reindex(new_index, copy=False)

            if this.dtype.kind == "M" and other.dtype.kind != "M":
                other = to_datetime(other)

            this_values = this._values
            other_values = other._values

        this_mask = notna(this_values)
        if this_mask.all():
            # No nulls in self -> result is just self reindexed
            if new_index is self.index:
                return self.copy()
            return self._constructor(
                this_values, index=new_index, name=self.name
            ).__finalize__(self, method="combine_first")

        if isinstance(this_values, ExtensionArray):
            return this_values._combine_first(other_values, new_index, self)

        # Use putmask for efficient in-place-style null replacement
        result = this_values.copy()
        np.putmask(result, ~this_mask, other_values)

        return self._constructor(
            result, index=new_index, name=self.name
        ).__finalize__(self, method="combine_first")
