    def round(self, decimals: int = 0) -> Self:
        """
        Round each value in the RangeIndex to the given number of decimals.

        For a RangeIndex, rounding with decimals >= 0 is a no-op since all
        values are integers. Returns a RangeIndex when possible.

        Parameters
        ----------
        decimals : int, optional
            Number of decimal places to round to. If decimals is negative,
            it specifies the number of positions to the left of the decimal point.

        Returns
        -------
        RangeIndex or Index
            A RangeIndex if decimals >= 0, otherwise an Index with rounded values.

        Examples
        --------
        >>> idx = pd.RangeIndex(0, 5, 1)
        >>> idx.round()
        RangeIndex(start=0, stop=5, step=1)

        >>> idx.round(decimals=-1)
        Index([0, 0, 0, 0, 0], dtype='int64')
        """
        if decimals >= 0:
            return self.copy()
        # For negative decimals, delegate to the base implementation
        return super().round(decimals=decimals)
