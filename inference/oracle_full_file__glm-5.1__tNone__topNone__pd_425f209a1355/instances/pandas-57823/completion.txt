    def argmin(self, axis=None, skipna=True, *args, **kwargs):
        """
        Return the index of the minimum value.

        For RangeIndex, the minimum value is always at position 0 if step > 0,
        or at position len(self)-1 if step < 0. If the index is empty,
        raises ValueError.

        Parameters
        ----------
        axis : {None}
            Unused. Accepted for compatibility with numpy.
        skipna : bool, default True
            Unused. Accepted for compatibility with numpy.
        *args, **kwargs
            Additional keywords have no effect but might be accepted for
            compatibility with numpy.

        Returns
        -------
        int
            Index of the minimum value.

        See Also
        --------
        Index.argmin : Return the index of the minimum value.
        numpy.ndarray.argmin : Equivalent method for numpy arrays.
        """
        nv.validate_argmin(args, kwargs)
        nv.validate_minmax_axis(axis)
        if not len(self):
            raise ValueError("attempt to get argmin of an empty sequence")
        if self.step > 0:
            return 0
        return len(self) - 1
