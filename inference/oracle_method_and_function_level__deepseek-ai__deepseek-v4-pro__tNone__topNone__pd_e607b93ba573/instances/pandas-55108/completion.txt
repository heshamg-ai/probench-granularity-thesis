    def _difference(self, other, sort):
        # overridden by RangeIndex

        this = self.unique()

        # Fast path: if other is a non-MultiIndex and this is a non-MultiIndex,
        # we can use libindex helpers or a mask-based approach that avoids
        # O(n log n) sorting when the indexes are disjoint.
        indexer = this.get_indexer_for(other)
        if is_object_dtype(this.dtype) and is_object_dtype(other.dtype):
            # For object dtype, get_indexer_for already has fast paths;
            # use the existing approach but avoid setdiff1d overhead when
            # indexes are disjoint (common case for .difference).
            mask = indexer == -1
            if mask.all():
                # Disjoint indexes — no overlaps, return all of `this`
                the_diff = this._values
                the_diff = _maybe_try_sort(the_diff, sort)
                return the_diff
            keep = np.flatnonzero(mask)
            if isinstance(this, ABCMultiIndex):
                the_diff = this.take(keep)
            else:
                the_diff = this._values.take(keep)
            the_diff = _maybe_try_sort(the_diff, sort)
            return the_diff

        # For non-object dtypes (e.g., DatetimeIndex, int64, etc.),
        # use boolean indexing which is faster than setdiff1d and
        # take/fancy indexing.
        mask = indexer == -1
        if mask.all():
            # Disjoint — no overlaps, return all of `this`
            the_diff = this._values
            the_diff = _maybe_try_sort(the_diff, sort)
            return the_diff

        keep = np.flatnonzero(mask)
        if isinstance(this, ABCMultiIndex):
            the_diff = this.take(keep)
        else:
            the_diff = this._values.take(keep)
        the_diff = _maybe_try_sort(the_diff, sort)
        return the_diff
