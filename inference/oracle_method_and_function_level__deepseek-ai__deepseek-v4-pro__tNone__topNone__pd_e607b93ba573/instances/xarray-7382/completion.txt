        # Index objects are the same; still must check index variables
        vars0 = elements[0][1]
        vars1 = elements[1][1]
        if vars0 is vars1:
            return True
        if len(vars0) != len(vars1):
            return False
        for k in vars0:
            if k not in vars1 or not vars0[k].equals(vars1[k]):
