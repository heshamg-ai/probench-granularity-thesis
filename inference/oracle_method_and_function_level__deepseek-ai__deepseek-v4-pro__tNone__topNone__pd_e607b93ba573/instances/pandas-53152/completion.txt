            stop = i - 1
            step = -1
        selected = pc.utf8_slice_codeunits(
            self._pa_array, start=start, stop=stop, step=step
        )
        # Use pc.if_else with null mask to avoid allocating a full-None array
        result = pc.if_else(
            pc.invert(out_of_bounds.fill_null(True)),
            selected,
            pa.nulls(self._pa_array.length(), type=self._pa_array.type),
        )
        return type(self)(result)
