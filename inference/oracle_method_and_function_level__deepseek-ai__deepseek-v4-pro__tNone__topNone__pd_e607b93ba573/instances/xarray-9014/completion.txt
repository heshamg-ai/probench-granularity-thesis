def _open_datatree_netcdf(
    ncDataset: ncDataset | ncDatasetLegacyH5,
    filename_or_obj: str | os.PathLike[Any] | BufferedIOBase | AbstractDataStore,
    **kwargs,
) -> DataTree:
    from xarray.backends.api import open_dataset
    from xarray.core.datatree import DataTree
    from xarray.core.treenode import NodePath

    # Extract decode_cf=kwargs if present, so we can handle it uniformly.
    decode_cf_arg = kwargs.pop("decode_cf", True)

    # Open the root dataset with decode_cf=True to get all string handling /
    # CF decoding applied consistently.
    ds = open_dataset(filename_or_obj, decode_cf=decode_cf_arg, **kwargs)
    tree_root = DataTree.from_dict({"/": ds})
    
    # Open all groups in a single pass, reusing the ncDataset handle.
    with ncDataset(filename_or_obj, mode="r") as ncds:
        # Pre-collect all group paths to avoid any iterator side-effects.
        group_paths = list(_iter_nc_groups(ncds))
    
    # Avoid re-encoding overhead by opening all subgroups with the same
    # decode_cf setting used for the root.
    for path in group_paths:
        subgroup_ds = open_dataset(
            filename_or_obj, group=path, decode_cf=decode_cf_arg, **kwargs
        )

        # TODO refactor to use __setitem__ once creation of new nodes by assigning Dataset works again
        node_name = NodePath(path).name
        new_node: DataTree = DataTree(name=node_name, data=subgroup_ds)
        tree_root._set_item(
            path,
            new_node,
            allow_overwrite=False,
            new_nodes_along_path=True,
        )
    return tree_root
