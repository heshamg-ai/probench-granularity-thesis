def _is_integral_type(arr):
    """Return True if the array has an integer dtype or contains only
    integral values (without iterating the whole array)."""
    if not hasattr(arr, 'dtype'):
        arr = np.asarray(arr)
    if np.issubdtype(arr.dtype, np.integer):
        return True
    # For object/string arrays, or lists, quick check first element
    if arr.size > 0:
        val = arr.flat[0]
        if isinstance(val, (int, np.integer)):
            return True
    return False
