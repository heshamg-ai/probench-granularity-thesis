    def append(self, other):
        """
        Append a collection of Index options together.

        Parameters
        ----------
        other : Index or list/tuple of indices

        Returns
        -------
        Index
            The combined index.

        Examples
        --------
        >>> mi = pd.MultiIndex.from_arrays([['a'], ['b']])
        >>> mi
        MultiIndex([('a', 'b')],
                   )
        >>> mi.append(mi)
        MultiIndex([('a', 'b'), ('a', 'b')],
                   )
        """
        if not isinstance(other, (list, tuple)):
            other = [other]

        if all(
            (isinstance(o, MultiIndex) and o.nlevels >= self.nlevels) for o in other
        ):
            # Pre-allocate arrays and build list comprehensions for speed
            nlevels = self.nlevels
            # Pre-compute level values for self once
            self_levels = [self._get_level_values(i) for i in range(nlevels)]

            # Pre-compute level values for all others (list of lists)
            other_levels = [
                [o._get_level_values(i) for o in other] for i in range(nlevels)
            ]

            # Fast-path: use ._concat_same_type for homogeneous, equal-dtype levels
            # across all indices being appended. Only applies when the Index
            # subclass / dtype matches across self and all others for each level.
            arrays = []
            names = []
            for i in range(nlevels):
                level_self = self_levels[i]
                level_others = other_levels[i]

                # Check if all others have the same type/dtype as level_self
                # to allow using the more efficient _concat_same_type path
                if all(
                    type(level_self) is type(o) and level_self.dtype == o.dtype
                    for o in level_others
                ):
                    combined = level_self._concat_same_type(level_others)
                else:
                    combined = level_self.append(level_others)

                arrays.append(combined)

                single_label_name = (
                    level_self.name == level_others[0].name
                    and all(level_self.name == x.name for x in level_others[1:])
                )
                names.append(level_self.name if single_label_name else None)

            return MultiIndex.from_arrays(arrays, names=names)

        to_concat = (self._values,) + tuple(k._values for k in other)
        new_tuples = np.concatenate(to_concat)

        # if all(isinstance(x, MultiIndex) for x in other):
        try:
            # We only get here if other contains at least one index with tuples,
            # setting names to None automatically
            return MultiIndex.from_tuples(new_tuples)
        except (TypeError, IndexError):
            return Index(new_tuples)
