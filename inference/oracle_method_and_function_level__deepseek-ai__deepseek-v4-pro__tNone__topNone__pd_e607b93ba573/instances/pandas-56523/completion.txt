def _is_monotonic_increasing(arr: np.ndarray) -> bool:
    """
    Check whether the given array is strictly or non-strictly monotonic increasing.
    Returns True for arrays with all equal values as well.
    """
    if len(arr) <= 1:
        return True
    # Use memcmp-style check for non-decreasing property
    # This avoids creating a large boolean array
    return not (arr[1:] < arr[:-1]).any()


def _sorted_join_indexers(
    left: np.ndarray,
    right: np.ndarray,
    how: JoinHow = "inner",
) -> tuple[npt.NDArray[np.intp], npt.NDArray[np.intp]]:
    """
    Compute join indexers for two sorted, monotonic, same-dtype arrays
    without hashing or factorization.

    Parameters
    ----------
    left : np.ndarray
        Sorted monotonic increasing values.
    right : np.ndarray
        Sorted monotonic increasing values.
    how : {'inner', 'outer', 'left', 'right'}

    Returns
    -------
    left_indexer : np.ndarray[np.intp]
    right_indexer : np.ndarray[np.intp]
    """
    # Use searchsorted to find matching positions.
    # For each key in the smaller array, find where it would be inserted
    # in the larger array, then walk both sides.

    if how == "inner":
        return _sorted_inner_join(left, right)
    elif how == "left":
        return _sorted_left_join(left, right)
    elif how == "right":
        right_idx, left_idx = _sorted_left_join(right, left)
        return left_idx, right_idx
    elif how == "outer":
        return _sorted_outer_join(left, right)
    else:
        raise ValueError(f"Unsupported join how: {how}")


def _sorted_inner_join(
    left: np.ndarray, right: np.ndarray
) -> tuple[npt.NDArray[np.intp], npt.NDArray[np.intp]]:
    """
    Inner join for sorted monotonic arrays.
    For each element in the smaller array, find all matching elements
    in the larger array using searchsorted to bound the range.
    """
    n_left = len(left)
    n_right = len(right)

    if n_left == 0 or n_right == 0:
        return np.array([], dtype=np.intp), np.array([], dtype=np.intp)

    # Walk both arrays simultaneously
    li = 0
    ri = 0
    left_indices = []
    right_indices = []

    while li < n_left and ri < n_right:
        lval = left[li]
        rval = right[ri]

        if lval < rval:
            li += 1
        elif rval < lval:
            ri += 1
        else:
            # Found matching value: count duplicates on both sides
            # Advance left through duplicates
            l_start = li
            while li < n_left and left[li] == lval:
                li += 1
            l_count = li - l_start

            # Advance right through duplicates
            r_start = ri
            while ri < n_right and right[ri] == rval:
                ri += 1
            r_count = ri - r_start

            # Cross product of duplicates
            for l_idx in range(l_start, li):
                for r_idx in range(r_start, ri):
                    left_indices.append(l_idx)
                    right_indices.append(r_idx)

    if not left_indices:
        return np.array([], dtype=np.intp), np.array([], dtype=np.intp)

    return np.array(left_indices, dtype=np.intp), np.array(right_indices, dtype=np.intp)


def _sorted_left_join(
    left: np.ndarray, right: np.ndarray
) -> tuple[npt.NDArray[np.intp], npt.NDArray[np.intp]]:
    """
    Left outer join for sorted monotonic arrays.
    All rows from left are preserved; for unmatched left rows,
    the right indexer is -1.
    """
    n_left = len(left)
    n_right = len(right)

    if n_left == 0:
        return np.array([], dtype=np.intp), np.array([], dtype=np.intp)

    if n_right == 0:
        return np.arange(n_left, dtype=np.intp), np.full(n_left, -1, dtype=np.intp)

    li = 0
    ri = 0
    left_indices = []
    right_indices = []

    while li < n_left and ri < n_right:
        lval = left[li]
        rval = right[ri]

        if lval < rval:
            # No match for this left value
            left_indices.append(li)
            right_indices.append(-1)
            li += 1
        elif rval < lval:
            ri += 1
        else:
            # Matching value: cross product
            l_start = li
            while li < n_left and left[li] == lval:
                li += 1

            r_start = ri
            while ri < n_right and right[ri] == rval:
                ri += 1

            for l_idx in range(l_start, li):
                for r_idx in range(r_start, ri):
                    left_indices.append(l_idx)
                    right_indices.append(r_idx)

    # Remaining left items have no match
    while li < n_left:
        left_indices.append(li)
        right_indices.append(-1)
        li += 1

    return np.array(left_indices, dtype=np.intp), np.array(right_indices, dtype=np.intp)


def _sorted_outer_join(

    Full outer join for sorted monotonic arrays.
    """
    n_left = len(left)
    n_right = len(right)

    if n_left == 0:
        return (
            np.full(n_right, -1, dtype=np.intp),
            np.arange(n_right, dtype=np.intp),
        )
    if n_right == 0:
        return (
            np.arange(n_left, dtype=np.intp),
            np.full(n_left, -1, dtype=np.intp),
        )

    li = 0
    ri = 0
    left_indices = []
    right_indices = []

    while li < n_left and ri < n_right:
        lval = left[li]
        rval = right[ri]

        if lval < rval:
            left_indices.append(li)
            right_indices.append(-1)
            li += 1
        elif rval < lval:
            left_indices.append(-1)
            right_indices.append(ri)
            ri += 1
        else:

            l_start = li
            while li < n_left and left[li] == lval:
                li += 1

            r_start = ri
            while ri < n_right and right[ri] == rval:
                ri += 1

            for l_idx in range(l_start, li):

                    left_indices.append(l_idx)
                    right_indices.append(r_idx)

    while li < n_left:
        left_indices.append(li)
        right_indices.append(-1)
        li += 1

    # Remaining right
    while ri < n_right:
        left_indices.append(-1)
        right_indices.append(ri)
        ri += 1
