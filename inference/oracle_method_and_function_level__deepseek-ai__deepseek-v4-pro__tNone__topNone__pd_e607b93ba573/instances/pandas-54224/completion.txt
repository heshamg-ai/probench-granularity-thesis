def transpose_homogeneous_pyarrow(
    arrow_arrays: Sequence["ArrowExtensionArray"],
) -> list["ArrowExtensionArray"]:
    """Transpose PyArrow-backed arrays in a list, but faster.

    Input should be a list of 1-dim ArrowExtensionArrays of equal length and all have
    the same dtype. The caller is responsible for ensuring validity of input data.
    """
    import pyarrow as pa

    chunks = [arr._pa_array.combine_chunks() for arr in arrow_arrays]
    table = pa.table({str(i): col for i, col in enumerate(chunks)})
    transposed_table = table.transpose()
    dtype = arrow_arrays[0].dtype
    arr_type = dtype.construct_array_type()
    result = [
        arr_type(transposed_table.column(i).combine_chunks())
        for i in range(transposed_table.num_columns)
    ]
    return result
