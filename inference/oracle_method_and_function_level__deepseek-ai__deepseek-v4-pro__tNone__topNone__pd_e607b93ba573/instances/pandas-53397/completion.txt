def _range_take(rng: range, indices: np.ndarray):
    """
    Take elements from a range using integer indices.

    Returns an Index (RangeIndex if possible, else Int64Index) or None
    if the fast path cannot be applied.
    """
    if indices.size == 0:
        return RangeIndex._simple_new(range(0), name=None)

    if indices.min() < -len(rng) or indices.max() >= len(rng):
        return None

    start, stop, step = rng.start, rng.stop, rng.step

    if step == 1:
        # monotonic increasing, no duplicates unless indices repeat
        take_vals = indices.astype(np.intp, copy=False)
        new_vals = start + take_vals
        lo = new_vals.min()
        hi = new_vals.max()
        if hi - lo + 1 <= 2 * indices.size:
            # dense enough: keep as RangeIndex
            return RangeIndex(lo, hi + 1, name=None).take(
                new_vals - lo, allow_fill=False
            )
        return RangeIndex._simple_new(new_vals, name=None)

    # step != 1 -> need full materialization
    vals = np.arange(start, stop, step)
    return RangeIndex._simple_new(vals[indices], name=None)
