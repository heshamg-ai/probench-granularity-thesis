def _round_field(values, name, freq):
    """Round datetime-like values to the specified frequency.

    Uses CFTimeIndex for cftime values and pandas Series `.dt` accessor
    for numpy datetime64 values.
    """
    if is_np_datetime_like(values.dtype):
        return _round_through_series(values, name, freq)
    return _round_through_cftimeindex(values, name, freq)


def _round_through_series(values, name, freq):
    """Round numpy datetime64 values via pandas Series `.dt` accessor."""
    import pandas as pd

    return getattr(pd.Series(values.ravel()).dt, name)(freq).values.reshape(values.shape)
