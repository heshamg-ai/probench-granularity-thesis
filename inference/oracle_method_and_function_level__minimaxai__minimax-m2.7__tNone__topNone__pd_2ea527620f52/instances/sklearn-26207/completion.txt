def _rescale_data(X, y, sample_weight):
    """Rescale data sample-wise by square root of sample_weight.

    For many linear models, this enables easy support for sample_weight because

        (y - X w)' S (y - X w)

    with S = diag(sample_weight) becomes

        ||y_rescaled - X_rescaled w||_2^2

    when setting

        y_rescaled = sqrt(S) y
        X_rescaled = sqrt(S) X

    Returns
    -------
    X_rescaled : {array-like, sparse matrix}

    y_rescaled : {array-like, sparse matrix}
    """
    n_samples = X.shape[0]
    sample_weight = np.asarray(sample_weight)
    if sample_weight.ndim == 0:
        sample_weight = np.full(n_samples, sample_weight, dtype=sample_weight.dtype)
    sample_weight_sqrt = np.sqrt(sample_weight)
    sw_matrix = sparse.dia_matrix((sample_weight_sqrt, 0), shape=(n_samples, n_samples))
    X = safe_sparse_dot(sw_matrix, X)
    y = safe_sparse_dot(sw_matrix, y)
    return X, y, sample_weight_sqrt

def _rescale_data(X, y, sample_weight):
    """Rescale data sample-wise by square root of sample_weight.

    For many linear models, this enables easy support for sample_weight because

        (y - X w)' S (y - X w)

    with S = diag(sample_weight) becomes

        ||y_rescaled - X_rescaled w||_2^2

    when setting

        y_rescaled = sqrt(S) y
        X_rescaled = sqrt(S) X

    Returns
    -------
    X_rescaled : {array-like, sparse matrix}

    y_rescaled : {array-like, sparse matrix}
    """
    n_samples = X.shape[0]
    sample_weight = np.asarray(sample_weight)
    if sample_weight.ndim == 0:
        sample_weight = np.full(n_samples, sample_weight, dtype=sample_weight.dtype)
    sample_weight_sqrt = np.sqrt(sample_weight)

    if sp.issparse(X):
        if sp.issparse(y):
            # Scale sparse y directly
            y = y.tocsc()
            y_data = y.data
            w = sample_weight_sqrt[y.indices]
            y_data *= w
        else:
            # For dense y, use dia_matrix (no copy made, just scale)
            sw_matrix = sparse.dia_matrix(
                (sample_weight_sqrt, 0), shape=(n_samples, n_samples)
            )
            y = safe_sparse_dot(sw_matrix, y)

        # Scale sparse X by rows directly (avoids constructing dia_matrix)
        X = X.tocsr()
        X_data = X.data
        # Repeat weights per non-zero element per row
        w = np.repeat(sample_weight_sqrt, np.diff(X.indptr))
        X_data *= w
    else:
        sw_matrix = sparse.dia_matrix(
            (sample_weight_sqrt, 0), shape=(n_samples, n_samples)
        )
        X = safe_sparse_dot(sw_matrix, X)
        if sp.issparse(y):
            # Scale sparse y directly
            y = y.tocsc()
            y_data = y.data
            w = sample_weight_sqrt[y.indices]
            y_data *= w
        else:
            y = safe_sparse_dot(sw_matrix, y)

    return X, y, sample_weight_sqrt
