        elif (
            self._is_homogeneous_type
            and dtypes
            and isinstance(dtypes[0], ExtensionDtype)
        ):
            # We have EAs with the same dtype. We can preserve that dtype in transpose.
            dtype = dtypes[0]

            # Check if all columns are masked extension arrays
            from pandas.core.arrays.masked import transpose_homogenous_masked_arrays

            if isinstance(dtype, ExtensionDtype) and dtype.name in (
                "Int8",
                "Int16",
                "Int32",
                "Int64",
                "UInt8",
                "UInt16",
                "UInt32",
                "UInt64",
                "Float32",
                "Float64",
            ):
                # Use optimized transpose for masked extension arrays
                new_values, new_index, new_columns = transpose_homogenous_masked_arrays(

                    new_values, index=new_index, columns=new_columns
