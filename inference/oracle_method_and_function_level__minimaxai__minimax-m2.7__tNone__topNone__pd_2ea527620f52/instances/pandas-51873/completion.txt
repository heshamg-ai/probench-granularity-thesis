        codes = codes or self.codes

        if len(levels) != len(codes):

            if len(level_codes) != codes_length:

                )
            if len(level_codes):
                level_len = len(level)
                min_val = level_codes.min()
                max_val = level_codes.max()
                if max_val >= level_len:
                    raise ValueError(
                        f"On level {i}, code max ({max_val}) >= length of "
                        f"level ({level_len}). NOTE: this index is in an "
                        "inconsistent state"
                    )
                if min_val < -1:
                    raise ValueError(f"On level {i}, code value ({min_val}) < -1")

            if len(level) != len(set(level)):
                raise ValueError(
                    f"Level values must be unique: {list(level)} on level {i}"
                )

            if self.sortorder > _lexsort_depth(self.codes, self.nlevels):
