        result = Dataset(dataset, coords=self.obj.coords, attrs=attrs)
        # Optimize: skip isel when stride is 1, since it would be a no-op
        if all(s == 1 for s in strides):
            return result
        return result.isel({d: slice(None, None, s) for d, s in zip(self.dim, strides)})
