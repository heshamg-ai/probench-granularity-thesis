    """
    Retrieves the index of the first valid value.

    Parameters
    ----------
    values : ndarray or ExtensionArray
    how : {'first', 'last'}

    is_valid: np.ndarray
        Mask to find na_values.

    Returns
    -------
    int or None
    """
    assert how in ["first", "last"]

    if len(values) == 0:  # early stop
        return None

    if values.ndim == 2:
        is_valid = is_valid.any(axis=1)  # reduce axis 1
        # For 2D, if it came from an ExtensionArray, it's now numpy

    # we can sometimes avoid conversion overhead
    if how == "first":
        idxpos = is_valid.argmax()

    elif how == "last":
        idxpos = len(values) - 1 - is_valid[::-1].argmax()

    # Use direct indexing; for numpy arrays this is fastest
    # For ExtensionArrays, the is_valid[idxpos] check ensures
    # we don't access out of bounds when all values are invalid

        return None
    # Incompatible return value type (got "signedinteger[Any]",
