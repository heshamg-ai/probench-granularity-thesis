def _are_keys_sorted(keys: list[ArrayLike]) -> bool:
    """
    Check if all key arrays are monotonically increasing.
    
    Returns True only if all keys are monotonically increasing.
    """
    for key in keys:
        values = np.asarray(key)
        if len(values) > 1 and not isinstance(values, ExtensionArray):
            if not is_list_like(values):
                continue
            # Check monotonicity using numpy
            try:
                if not (np.diff(values) >= 0).all():
                    return False
            except (TypeError, ValueError):
                # Cannot compare values (e.g., mixed types)
                return False
    return True
