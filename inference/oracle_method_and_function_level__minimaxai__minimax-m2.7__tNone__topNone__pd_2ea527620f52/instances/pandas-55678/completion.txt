                # Fast path: if dtypes match exactly, use values directly
                if lbv.dtype == rbv.dtype:
                    # For i8 types (datetime64), convert to int64 view
                    if needs_i8_conversion(lbv.dtype):
                        lbv = lbv.view("i8")
                        rbv = rbv.view("i8")
                    # else: use as-is (same dtype, no conversion needed)
                else:
                    # Different dtypes: find common dtype efficiently
                    # Check if both need i8 conversion (datetime types)
                    lbv_i8 = needs_i8_conversion(lbv.dtype)
                    rbv_i8 = needs_i8_conversion(rbv.dtype)
                    if lbv_i8 and rbv_i8:
                        # Both datetime-like: convert both to int64
                        lbv = lbv.view("i8")
                        rbv = rbv.view("i8")
                    elif lbv_i8 or rbv_i8:
                        # One datetime, one not - incompatible, will error later
                        if lbv_i8:
                            lbv = lbv.view("i8")
                    else:
                        # Neither datetime: find common dtype and cast
                        try:
                            common_dtype = np.result_type(lbv.dtype, rbv.dtype)
                            lbv = lbv.astype(common_dtype, copy=False)
                            rbv = rbv.astype(common_dtype, copy=False)
                        except (TypeError, ValueError):
                            # Fall back to object dtype for incompatible types
                            lbv = lbv.astype(object)
                            rbv = rbv.astype(object)
