    def _compute_ridx_for_left_join(
        self, other_start: int, other_step: int, other_len: int
    ) -> npt.NDArray[np.intp]:
        """Compute right indexer for left join with another RangeIndex."""
        self_start = self._range.start
        self_step = self._range.step
        self_len = len(self._range)

        ridx = np.zeros(self_len, dtype=np.intp)
        for i in range(self_len):
            val = self_start + i * self_step
            if other_start <= val < other_start + other_len * other_step:
                if (val - other_start) % other_step == 0:
                    ridx[i] = (val - other_start) // other_step
                else:
                    ridx[i] = 0
            else:
                ridx[i] = -1
        return ridx
