The main bottlenecks are:
1. `fillna` on the entire condition DataFrame
2. `infer_objects` conversion 
3. Per-column dtype checking loop
4. Unnecessary reindex when columns are already aligned

                cond = cond.fillna(fill_value)
                cond = cond.infer_objects()
            else:
                # Fast path: check if all columns are already boolean and aligned
                all_bool = all(is_bool_dtype(_dt) for _dt in cond.dtypes)
                same_columns = cond.columns.equals(self.columns)
                same_index = cond.index.equals(self.index)

                if all_bool and not cond.isna().any().any() and same_columns and same_index:
                    # Already properly formatted, skip expensive operations
                    pass
                else:
                    # Need to process: fill NA, check dtypes, align
                    cond = cond.fillna(fill_value)
                    cond = cond.infer_objects()

                    for _dt in cond.dtypes:
                        if not is_bool_dtype(_dt):
                            raise TypeError(msg.format(dtype=_dt))

                    if cond._mgr.any_extension_types:
                        # GH51574: avoid object ndarray conversion later on
                        cond = cond._constructor(
                            cond.to_numpy(dtype=bool, na_value=fill_value),
