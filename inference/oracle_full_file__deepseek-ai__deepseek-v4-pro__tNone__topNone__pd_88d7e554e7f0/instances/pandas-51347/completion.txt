            or pa.types.is_duration(pa_type)

            result[self.isna()] = na_value
        return result

        copy: bool = False,
        na_value: object = lib.no_default,
    ) -> np.ndarray:
        if dtype is None and self._hasna:
            dtype = object
        if na_value is lib.no_default:
            na_value = self.dtype.na_value

        if (
            is_object_dtype(dtype)

            or pa.types.is_duration(pa_type)
        ):
            # Pre-size the result array and fill it with values,
            # only converting to list if necessary for the PyArrow conversion
            length = len(self)
            result = np.empty(length, dtype=dtype)

            chunks = list(self._data.iterchunks())
            if chunks:
                # Use pyarrow's fast columnar to_pylist for object dtype,
                # or to_numpy for other dtypes with no nulls

                    # methods where possible, falling back to.to_numpy for arrays
                    # without nulls.

                        # For object output with nulls, we use a two-step approach:

                        if is_object_dtype(dtype):

                        else:
                            result = np.array(arr, dtype=dtype, copy=copy)
                else:
