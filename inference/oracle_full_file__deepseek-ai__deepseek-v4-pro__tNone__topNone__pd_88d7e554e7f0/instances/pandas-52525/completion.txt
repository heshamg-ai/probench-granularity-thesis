            return result
        elif self._hasna:
            null_count = self._pa_array.null_count
            if (
                not copy
                and null_count > 0
                and dtype is not None
                and not isna(na_value)
                and issubclass(np.dtype(dtype).type, np.floating)
                and np.isnan(na_value)
            ):
                # Fast path: cast valid data to float, then set NaN for nulls
                result = self._pa_array.to_numpy(zero_copy_only=False)
                result = result.astype(dtype, copy=False)
                if null_count == len(self):
                    result[:] = np.nan
                else:
                    result[self.isna()] = np.nan
                return result
            result = self._pa_array.to_numpy(zero_copy_only=False)
            if dtype is not None:
                result = result.astype(dtype, copy=False)
            result[self.isna()] = na_value
            if not copy:
                return result
            return result.copy()
        else:
            result = np.asarray(self._pa_array, dtype=dtype)
            if copy:
                result = result.copy()
        if self._hasna:
            result[self.isna()] = na_value
        return result
