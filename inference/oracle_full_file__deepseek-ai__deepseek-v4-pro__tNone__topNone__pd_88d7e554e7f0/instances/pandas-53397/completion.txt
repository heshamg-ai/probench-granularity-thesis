        **kwargs,

        if is_scalar(indices):
            raise TypeError("Expected indices to be array-like")
        indices = ensure_platform_int(indices)

        allow_fill = self._maybe_disallow_fill(allow_fill, fill_value, indices)

        if allow_fill:
            # RangeIndex cannot hold NA
            raise ValueError(
                "Unable to fill values because RangeIndex cannot contain NA"
            )

            # Fast path: use arithmetic to compute values from RangeIndex
            taken_start = self._range.start
            taken_step = self._range.step
            # Compute result as RangeIndex if step != 0;
            # fallback for empty/single element cases
            taken = taken_start + indices * taken_step
        except TypeError:
            # Could happen with weird indexer types; fallback to materializing

        return self._shallow_copy(taken, name=self.name)
