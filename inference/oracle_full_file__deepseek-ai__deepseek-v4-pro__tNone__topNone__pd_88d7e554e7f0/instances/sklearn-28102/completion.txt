    if len(distinct_values) <= max_bins:
        midpoints = distinct_values[:-1] + distinct_values[1:]
        midpoints *= 0.5
    else:
        # Avoid expensive full sort by computing approximate quantile
        # midpoints from the unique values and their counts.  This is a
        # large speed and memory win on data with many repeated values,
        # and still accurate enough for binning.
        distinct_values, counts = np.unique(col_data, return_counts=True)
        cum_counts = np.cumsum(counts, dtype=np.float64)
        total = cum_counts[-1]
        # quantile positions between 0 and total for the interior boundaries
        step = total / max_bins
        q_positions = np.arange(1, max_bins) * step
        idx = np.searchsorted(cum_counts, q_positions, side="right")
        idx = np.clip(idx, 0, len(distinct_values) - 1)
        midpoints = (distinct_values[idx - 1] + distinct_values[idx]) * 0.5
        midpoints = midpoints.astype(X_DTYPE)
