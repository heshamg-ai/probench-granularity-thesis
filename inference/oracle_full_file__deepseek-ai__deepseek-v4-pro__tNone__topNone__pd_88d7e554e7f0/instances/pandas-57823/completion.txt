    def argmin(self, axis=None, skipna: bool = True, *args, **kwargs) -> int:
        """
        Return int position of the smallest value in the Index.

        It is always 0 for a monotonically increasing RangeIndex,
        and len(self) - 1 for a monotonically decreasing RangeIndex.

        Parameters
        ----------
        axis : None
            Dummy argument to keep compatibility with Index.
        skipna : bool, default True
            Dummy argument to keep compatibility with Index.
        *args, **kwargs
            Dummy argument to keep compatibility with Index.

        Returns
        -------
        int
            Position of the minimum (0 for increasing, len-1 for decreasing).
        """
        nv.validate_minmax_axis(axis)
        nv.validate_min(args, kwargs)
        if len(self) == 0:
            raise ValueError("attempt to get argmin of an empty sequence")
        return 0 if self._range.step > 0 else len(self) - 1
