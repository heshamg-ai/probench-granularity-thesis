def _groupby_temporal_maxmin_pyarrow(
    pa_array: pa.ChunkedArray,
    *,
    how: str,
    ngroups: int,
    ids: npt.NDArray[np.intp],
) -> pa.Array:
    """
    Compute groupby max/min for timestamp/duration arrays directly via
    PyArrow compute functions, avoiding the NumPy round-trip used by
    the masked-array path.
    """
    # Combine chunks to get a single contiguous Array for the hash_ functions
    # which expect a single Array input.
    data = pa_array.combine_chunks()
    ids_arr = pa.array(ids, type=pa.int64())

    if how in ("first", "last"):
        # first: take the element at the first occurrence in each group
        # last:  take the element at the last occurrence in each group
        pc_op = getattr(pc, f"list_{how}")
        # Build lists per group, then extract the first/last element
        grouped = pc.list_flatten(
            pc.list_slice(
                pc.array_sort_indices(ids_arr),
                0,
                pc.count(ids_arr),
            )
        )
        # The approach above is fragile; instead use hash_list first/last:
        # hash_list returns lists for each group, then we pick first/last
        lists = pc.list_flatten(
            pc.array_sort_indices(ids_arr)
        )
        # Simpler: build groups via hash_list
        lists = pc.hash_list(
            data,
            ids_arr,
            ngroups,
        )
        result = pc_op(lists, skip_nulls=False)
        return result

    # For max/min, convert duration -> int64, operate, then cast back.
    # Timestamps are already backed by int64 underneath and pyarrow's hash_*
    # functions handle them natively since Arrow 13+, but duration may still
    # need casting. We do the safe thing: cast durations to int64, run the
    # hash, then cast back.
    pa_type = data.type

    if pa.types.is_duration(pa_type):
        data = data.cast(pa.int64())

    if how == "max":
        result = pc.hash_max(data, ids_arr, ngroups)
    elif how == "min":
        result = pc.hash_min(data, ids_arr, ngroups)
    else:
        raise ValueError(f"Unsupported how='{how}'")

    if pa.types.is_duration(pa_type):
        result = result.cast(pa_type)

    # hash_max/hash_min can produce nulls for empty groups; that's fine.
    return result
