def _is_string_object_array(values: np.ndarray) -> bool:
    """
    Fast check if an object-dtype ndarray contains only strings.

    Avoids calling lib.infer_dtype (which scans the entire array and
    allocates intermediate objects) by sampling a few elements.

    Parameters
    ----------
    values : np.ndarray with dtype=object

    Returns
    -------
    bool
    """
    # For empty arrays, default to string (object hashtable would work too, but
    # string is faster and semantics are preserved).
    if values.size == 0:
        return True

    # Sample a few elements to avoid scanning the entire array
    # when the array is large and non-string.
    sample_size = min(100, values.size)
    stride = max(1, values.size // sample_size)
    sample_indices = slice(0, values.size, stride) if values.size > sample_size else ...
    sample = values[sample_indices]

    # Use a plain Python check for the sampled subset.
    # None is a valid string-like in pandas (becomes NA), but
    # lib.infer_dtype(skipna=False) classifies arrays with None as
    # "mixed" not "string". To match the original `lib.infer_dtype`
    # behavior, we must reject arrays with non-string, non-None objects.
    for v in sample:
        if v is None:
            continue
        if not isinstance(v, str):
            return False

    return True
