        labels_for_lexsort = np.where(ids == -1, na_label_for_sorting, ids)

        # Pre-allocate reusable output buffer for the common 1d case
        _out_buf = np.empty((ngroups, nqs), dtype=np.float64)
        _group_quantile = libgroupby.group_quantile

        def blk_func(values: ArrayLike) -> ArrayLike:
            orig_vals = values
            if isinstance(values, BaseMaskedArray):

            is_datetimelike = needs_i8_conversion(values.dtype)

            vals, inference = pre_processor(values)

            if vals.ndim == 1:
