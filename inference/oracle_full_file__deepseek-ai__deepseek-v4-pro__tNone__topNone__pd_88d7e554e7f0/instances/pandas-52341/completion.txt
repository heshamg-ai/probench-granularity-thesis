def nanany(
    values: np.ndarray,
    *,
    axis: AxisInt | None = None,
    skipna: bool = True,
    mask: npt.NDArray[np.bool_] | None = None,
) -> bool:
    """
    Check if any elements along an axis evaluate to True.

    Parameters
    ----------
    values : ndarray
    axis : int, optional
    skipna : bool, default True
    mask : ndarray[bool], optional
        nan-mask if known

    Returns
    -------
    result : bool

    Examples
    --------
    >>> from pandas.core import nanops
    >>> s = pd.Series([1, 2])
    >>> nanops.nanany(s.values)
    True

    >>> from pandas.core import nanops
    >>> s = pd.Series([np.nan])
    >>> nanops.nanany(s.values)
    False
    """
    if needs_i8_conversion(values.dtype) and values.dtype.kind != "m":
        # GH#34479
        warnings.warn(
            "'any' with datetime64 dtypes is deprecated and will raise in a "
            "future version. Use (obj != pd.Timestamp(0)).any() instead.",
            FutureWarning,
            stacklevel=find_stack_level(),
        )

    if is_object_dtype(values):
        if axis is None and not values.ndim > 1:
            if skipna and mask is None:
                mask = isna(values)
            if skipna and mask is not None:
                # Skip NaN values in object arrays
                for val in values.flat:
                    if val is not None and not (
                        isinstance(val, float) and np.isnan(val)
                    ):
                        if val:
                            return True
                return False
            else:
                return values.astype(bool).any(axis=axis)
        values = values.astype(bool)
    elif is_bool_dtype(values):
        if skipna and mask is None:
            mask = isna(values)
        if skipna and mask is not None and mask.any():
            # Fast path: if all values are NaN, return False
            if mask.all():
                return False
            # Use short-circuit evaluation for 1D bool arrays
            if axis is None and not values.ndim > 1:
                for i in range(len(values)):
                    if not mask[i] and values[i]:
                        return True
                return False
        return values.any(axis)
    elif values.dtype.kind in "iuf":
        # For numeric types, NaN is already handled if skipna is True
        return values.any(axis)

    values, _, _, _, _ = _get_values(values, skipna, fill_value=False, mask=mask)

    # For object type, any won't necessarily return
    # boolean values (numpy/numpy#4352)
    if is_object_dtype(values):
        values = values.astype(bool)

    # error: Incompatible return value type (got "Union[bool_, ndarray]", expected
    # "bool")
    return values.any(axis)  # type: ignore[return-value]
