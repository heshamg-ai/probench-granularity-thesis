        # This avoids redundant iteration and allows batch processing

        # Categorize variables before processing
        # Skip variables already in indexers
        to_interp: list[tuple[Hashable, Variable]] = []  # (name, var) for numeric
        reindex_vars: list[Hashable] = []  # names for non-numeric with interp dims
        keep_vars: list[tuple[Hashable, Variable]] = []  # (name, var) for independent

        for name, var in obj._variables.items():
            if name in indexer_names_set:
                continue

            var_dims = var.dims
            # Check if variable has any dimension in indexers
            has_interp_dim = bool(indexer_dims & var_dims)

            if not has_interp_dim:
                # Variable doesn't depend on any interpolated dimensions
                keep_vars.append((name, var))
            else:
                dtype_kind = var.dtype.kind
                if dtype_kind in "uifc":
                    to_interp.append((name, var))
                elif dtype_kind in "ObU":
                    reindex_vars.append(name)

        # Process variables
        variables: dict[Hashable, Variable] = {}

        # Handle numeric variables - use appropriate indexers

        # Handle non-interpolated variables directly
