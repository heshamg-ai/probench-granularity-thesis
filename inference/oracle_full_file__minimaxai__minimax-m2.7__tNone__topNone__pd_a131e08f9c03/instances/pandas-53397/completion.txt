    def _can_use_range_take(self, taken_values: np.ndarray) -> bool:
        """
        Check if taken_values represents a valid range (consecutive values with same step).
        """
        if len(taken_values) <= 1:
            return True
        # Check if values are consecutive with the same step
        diffs = np.diff(taken_values)
        return len(np.unique(diffs)) == 1 and diffs[0] == self.step
