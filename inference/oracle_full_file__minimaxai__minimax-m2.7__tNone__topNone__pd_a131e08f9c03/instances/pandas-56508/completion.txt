def hash_array(
    values: ArrayLike,
    encoding: str = "utf-8",
    hash_key: str = "012345678912345",
    categorize: bool = True,
) -> npt.NDArray[np.uint64]:
    """
    Return a uint64 ndarray of values of the same length as `values`.

    Parameters
    ----------
    values : ExtensionArray, Index, or ndarray
        Array of values to hash.
    encoding : str, default "utf-8"
        Encoding for string/float/object values.
    hash_key : str, default "012345678912345"
        Hash key for string/float/object values.
    categorize : bool, default True
        Whether to first categorize object arrays before hashing. This is more
        efficient when the array contains duplicate values.

    Returns
    -------
    np.ndarray[uint64]

    Notes
    -----
    For the hash to be consistent across all pandas versions, the hash for
    a NA value (NaT, None, np.nan) has to be pre-generated via
    ``vectorized_hash`` (or ``hash_array`` with an array of all identical
    values that contain the NA, e.g. ``np.array([NaT], dtype=object)``).
    """
    if not hasattr(values, "dtype"):
        values = np.asarray(values)

    if isinstance(values.dtype, BaseMaskedDtype):
        # Avoid hashing the mask together with the data, which would cause
        #  collisions between e.g. [1, NA] and [NA, 1]
        from pandas.core.arrays.base import ExtensionArray

        if not isinstance(values, ExtensionArray):
            values = values._data

        # Hashing: BaseMaskedDtype -> uint64
        #
        # We use two independent 64-bit hash values. The first is
        # hash(0) combined with the address of the mask. The second is
        # hash(1) combined with the address of the data. We XOR
        # these together to avoid symmetry across (data, mask) swap.
        # See BaseMaskedDtype.equals for similar logic.
        addr_mask = hash(
            id(values._mask),
            encoding=encoding,
            hash_key=hash_key,
        )
        addr_data = hash(
            id(values._data),
            encoding=encoding,
            hash_key=hash_key,
        )
        return addr_data ^ addr_mask

    elif hasattr(values, "dtype"):

        if values.dtype.kind == "f":
            values = np.asarray(values, dtype=float)

        elif values.dtype.kind == "b":
            values = np.asarray(values, dtype=bool)

        elif values.dtype.kind == "c":
            values = np.asarray(values, dtype=complex)

        elif values.dtype == object:
            # Current behavior is to hash object arrays to uint64
            #  (e.g. datetimetz, naive datetime64, PeriodDtype)
            values = np.asarray(values, dtype=object)

        elif lib.is_categorical_dtype(values.dtype):
            values = np.asarray(values)

        elif lib.is_period_dtype(values.dtype):
            values = np.asarray(values)

        elif lib.is_interval_dtype(values.dtype):
            values = np.asarray(values)

        else:
            raise NotImplementedError(f"{values.dtype} type not accounted for")

        return hash_array(
            values,
            encoding=encoding,
            hash_key=hash_key,
            categorize=categorize,
        )
