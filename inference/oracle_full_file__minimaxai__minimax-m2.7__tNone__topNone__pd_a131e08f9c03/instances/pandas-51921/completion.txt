    Parameters
    ----------
    values : ndarray

    Returns

    Check if we can use string hashtable instead of object hashtable.

    Parameters
    ----------
    values : ndarray

    Returns

    str
    """
    ndtype = values.dtype.name
    if ndtype == "object":
        # it's cheaper to use a String Hash Table than Object; we infer
        # including nulls because that is the only difference between
        # StringHashTable and ObjectHashtable
        # Optimization: check first element to determine if it could be a string
        # This is much faster than infer_dtype for arrays that are clearly not strings
        if len(values) > 0:
            first_val = values[0]
            if isinstance(first_val, str):
                # First element is a string, need to verify all are strings
