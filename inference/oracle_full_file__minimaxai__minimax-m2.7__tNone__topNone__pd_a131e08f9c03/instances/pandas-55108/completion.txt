    def _difference(self, other, sort):
        # overridden by RangeIndex

        this = self.unique()

        # Fast path for disjoint indexes: check if ranges overlap
        if (
            isinstance(self, ABCDatetimeIndex)
            and isinstance(other, ABCDatetimeIndex)
            and len(self) > 0
            and len(other) > 0
        ):
            self_min = self.min()
            self_max = self.max()
            other_min = other.min()
            other_max = other.max()
            # If ranges are disjoint, no elements can be in common
            if other_min > self_max or other_max < self_min:
                the_diff = this._values
                return _maybe_try_sort(the_diff, sort)

        indexer = this.get_indexer_for(other)
        indexer = indexer.take((indexer != -1).nonzero()[0])
