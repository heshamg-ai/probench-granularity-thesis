    def update_dtype(self, dtype: str_type | CategoricalDtype) -> CategoricalDtype:
        """
        Returns a CategoricalDtype with categories and ordered taken from dtype
        if specified, otherwise falling back to self if unspecified

        Parameters
        ----------
        dtype : CategoricalDtype

        Returns
        -------
        new_dtype : CategoricalDtype
        """
        if isinstance(dtype, str) and dtype == "category":
            # dtype='category' should not change anything
            return self
        elif not self.is_dtype(dtype):
            raise ValueError(
                f"a CategoricalDtype must be passed to perform an update, "
                f"got {dtype!r}"
            )
        else:
            # from here on, dtype is a CategoricalDtype
            dtype = cast(CategoricalDtype, dtype)

        # update categories/ordered unless they've been explicitly passed as None
        new_categories = (
            dtype.categories if dtype.categories is not None else self.categories
        )
        new_ordered = dtype.ordered if dtype.ordered is not None else self.ordered

        return CategoricalDtype(new_categories, new_ordered)

    def update_dtype(self, dtype: str_type | CategoricalDtype) -> CategoricalDtype:
        """
        Returns a CategoricalDtype with categories and ordered taken from dtype
        if specified, otherwise falling back to self if unspecified

        Parameters
        ----------
        dtype : CategoricalDtype

        Returns
        -------
        new_dtype : CategoricalDtype
        """
        if isinstance(dtype, str) and dtype == "category":
            # dtype='category' should not change anything
            return self
        elif not self.is_dtype(dtype):
            raise ValueError(
                f"a CategoricalDtype must be passed to perform an update, "
                f"got {dtype!r}"
            )
        else:
            # from here on, dtype is a CategoricalDtype
            dtype = cast(CategoricalDtype, dtype)

        # Fast path: both categories and ordered are specified
        # Use direct attribute access to avoid property overhead
        dtype_cats = dtype._categories
        dtype_ord = dtype._ordered
        if dtype_cats is not None and dtype_ord is not None:
            # Both are specified - skip validation using _from_fastpath
            # Return self if nothing would change
            if (
                dtype_cats is self._categories
                or dtype_cats == self._categories
            ) and dtype_ord == self._ordered:
                return self
            return CategoricalDtype._from_fastpath(dtype_cats, dtype_ord)

        # Slower path: at least one is None, need to merge with self
        new_categories = dtype_cats if dtype_cats is not None else self._categories
        new_ordered = dtype_ord if dtype_ord is not None else self._ordered

        return CategoricalDtype(new_categories, new_ordered)
