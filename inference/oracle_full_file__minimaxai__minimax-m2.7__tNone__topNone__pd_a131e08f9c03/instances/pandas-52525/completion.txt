

Looking at the current `to_numpy` implementation, I can identify several optimization opportunities:

1. When converting to float with `na_value=np.nan` and the array has nulls, the current code creates a copy, sets values, then converts to numpy - this is inefficient
2. For integer types converted to float, we should cast to float64 first to support NaN, then use optimized path
3. Pre-allocating the result array with np.nan and filling only non-null positions avoids intermediate copies

Let me implement the optimization:

            result[self.isna()] = na_value
        return result

[REPLACE]
    @doc(ExtensionArray.to_numpy)
    def to_numpy(
        self,
        dtype: npt.DTypeLike | None = None,
        copy: bool = False,
        na_value: object = lib.no_default,
    ) -> np.ndarray:
        if dtype is None and self._hasna:
            dtype = object
        if na_value is lib.no_default:
            na_value = self.dtype.na_value

        pa_type = self._pa_array.type
        if pa.types.is_temporal(pa_type) and not pa.types.is_date(pa_type):
            # temporal types with units and/or timezones currently
            #  require pandas/python scalars to pass all tests
            # TODO: improve performance (this is slow)
            result = np.array(list(self), dtype=dtype)
        elif is_object_dtype(dtype) and self._hasna:
            result = np.empty(len(self), dtype=object)
            mask = ~self.isna()
            result[mask] = np.asarray(self[mask]._pa_array)
        elif pa.types.is_null(self._pa_array.type):
            result = np.asarray(self._pa_array, dtype=dtype)
            if not isna(na_value):
                result[:] = na_value
            return result
        elif (
            (dtype is None or pa.types.is_floating(pa_type))
            and self._hasna
            and na_value is np.nan
        ):
            # Fast path for float-compatible types with np.nan as na_value
            # Pre-allocate result array with np.nan, fill only non-null positions
            if pa.types.is_floating(pa_type):
                pa_float_type = pa_type
            elif pa.types.is_integer(pa_type):
                pa_float_type = pa.float64()
            else:
                pa_float_type = pa.float64()

            data = self._pa_array
            hasna = self._hasna
            if not hasna:
                result = np.asarray(data, dtype=dtype)
                if copy:
                    result = result.copy()
                return result

            null_count = self._pa_array.null_count
            if null_count == len(self):
                return np.full(len(self), np.nan, dtype=np.float64)

            mask = self._pa_array.is_null().to_numpy()
            valid_indices = np.where(~mask)[0]

            if pa_float_type != pa_type:
                data = data.cast(pa_float_type)

            result = np.full(len(self), np.nan, dtype=np.float64)

            data_combined = data.combine_chunks()
            non_null_data = data_combined.to_numpy(zero_copy_only=False)
            result[valid_indices] = non_null_data[~mask]

            return result
        elif self._hasna:
            data = self.copy()
            data[self.isna()] = na_value
            return np.asarray(data._pa_array, dtype=dtype)
        else:
            result = np.asarray(self._pa_array, dtype=dtype)
            if copy:
                result = result.copy()
        if self._hasna:
            result[self.isna()] = na_value
        return result
[/REPLACE]
