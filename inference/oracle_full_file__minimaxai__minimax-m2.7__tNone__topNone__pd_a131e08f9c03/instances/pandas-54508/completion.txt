    def _get_value(self, row, col, takeable=False):
        """
        Quickly retrieve single value at passed column and index.

        Uses a fast-path if `takeable` is True.

        Parameters
        ----------
        row : int
            Integer row label
        col : int, str
            Column label
        takeable : bool
            Whether the index in `data` can be taken directly.
            If True, `row` refers to position in `data`,
            if False, `row` refers to label in `index`.

        Returns
        -------
        scalar
        """
        if takeable:
            # The fast-path: use underlying array directly
            # This avoids the overhead of converting indices
            if self._mgr.is_single_block:
                # Single block case - direct array access
                block = self._mgr.blocks[0]
                if block.is_extension:
                    # Extension dtype - use array directly
                    values = block.array_values
                    if values.ndim == 1:
                        return values[row]
                    else:
                        return values[row, col]
                # NumPy block case
                if block.values.ndim == 1:
                    return block.values[row]
                return block.values[row, col]
            
            # Multi-block case
            blknos = self._mgr.blknos
            blklocs = self._mgr.blklocs
            blkno = blknos[row]
            loc = blklocs[row]
            block = self._mgr.blocks[blkno]
            
            if block.is_extension:
                # Extension dtype case
                values = block.array_values
                if values.ndim == 1:
                    return values[loc]
                return values[loc, col] if col is not None else values[loc]
            
            return block.values[loc, col]
        
        # Slow path - index lookup
        index = self.index
        col = self.columns.get_loc(col)
        return self._get_value(row, col, takeable=True)

    def iget(self, key):
        """Perform the equivalent of 2D slicing for a single block.

        Parameters
        ----------
        key : tuple of int and/or slice objects, or a single slice object
            A 2D indexer with all keys in the same dimension as self.ndim.

        Returns
        -------
        array-like
        """
        if isinstance(key, tuple):
            # 2D indexing
            if len(key) == 2:
                row_sel, col_sel = key
                if isinstance(row_sel, slice) and row_sel == slice(None):
                    # Column selection from 2D block
                    if isinstance(col_sel, slice) and col_sel == slice(None):
                        return self.values
                    return self.values[:, col_sel]

                    return self.values[row_sel]
                else:
                    return self.values[row_sel, col_sel]
        # Single indexer
        return self.values[key]
