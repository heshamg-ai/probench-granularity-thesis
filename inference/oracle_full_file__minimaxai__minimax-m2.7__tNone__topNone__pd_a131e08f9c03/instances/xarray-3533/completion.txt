        **indexers_kwargs: Any,

        """Return a new DataArray whose data is given by integer indexing
        along the specified dimension(s).

        See Also
        --------
        Dataset.isel
        DataArray.sel
        """
        indexers = either_dict_or_kwargs(indexers, indexers_kwargs, "isel")
        
        # Fast path for simple integer indexing on small arrays
        # Check if all indexers are integers (or can be converted to integers)
        if not drop:
            all_int = True
            for k, v in indexers.items():
                if isinstance(v, integer_types):
                    pass  # already an integer
                elif isinstance(v, np.ndarray):
                    if v.ndim != 0 or v.dtype.kind not in 'iu':
                        all_int = False
                        break
                else:
                    try:
                        int(v)
                    except (TypeError, ValueError):
                        all_int = False
                        break
            
            if all_int and len(indexers) > 0:
                # All indexers are effectively integers - use optimized path
                # This avoids the overhead of converting to/from Dataset
                from .variable import Variable
                from .indexes import propagate_indexes
                
                # Get the indexed variable
                key = tuple(indexers.get(dim, slice(None)) for dim in self.dims)
                indexed_var = self.variable[key]
                
                # Determine which dimensions are being indexed away
                int_dims = set()
                for dim, idx in indexers.items():
                    if isinstance(idx, integer_types):
                        int_dims.add(dim)
                    elif isinstance(idx, np.ndarray) and idx.ndim == 0:
                        int_dims.add(dim)
                    else:
                        try:
                            int(idx)
                            int_dims.add(dim)
                        except (TypeError, ValueError):
                            pass
                
                # Build new coordinates
                new_coords = {}
                for name, coord_var in self._coords.items():
                    coord_dims = set(coord_var.dims)

                        # Need to check if we can extract from this coordinate
                        if coord_dims <= int_dims:

                    else:
                        # Coordinate doesn't depend on indexed dimensions
                        new_coords[name] = coord_var
                
                # Get new indexes
                new_indexes = propagate_indexes(self._indexes, exclude=int_dims)

                    variable=indexed_var,
                    coords=new_coords,
                    indexes=new_indexes,
