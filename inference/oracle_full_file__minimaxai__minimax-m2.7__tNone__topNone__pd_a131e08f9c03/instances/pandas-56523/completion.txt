def _is_sorted_monotonic(arr: ArrayLike) -> bool:
    """
    Check if array is sorted in ascending order.
    Returns True if arr[i] <= arr[i+1] for all valid i.
    """
    if len(arr) <= 1:
        return True
    if isinstance(arr, np.ndarray):
        return bool(np.all(arr[:-1] <= arr[1:]))
    # For ExtensionArrays, check manually
    for i in range(len(arr) - 1):
        lv = arr[i]
        rv = arr[i + 1]
        # Handle NA values - consider them as breaking sort for simplicity
        if isna(lv) or isna(rv):
            return False
        if lv > rv:
            return False
    return True


def _merge_join_indexer(
    left_vals: ArrayLike,
    right_vals: ArrayLike,
    how: JoinHow,
) -> tuple[npt.NDArray[np.intp], npt.NDArray[np.intp]]:
    """
    Optimized merge join for sorted monotonic keys.

    Uses a two-pointer algorithm which is O(n+m) and avoids hashing overhead.
    """
    n_left = len(left_vals)
    n_right = len(right_vals)

    # Estimate max result size
    if how == "inner":
        max_size = min(n_left, n_right)
    elif how in ("left", "right"):
        max_size = max(n_left, n_right)
    else:  # outer
        max_size = n_left + n_right

    left_indexer = np.empty(max_size, dtype=np.intp)
    right_indexer = np.empty(max_size, dtype=np.intp)
    pos = 0

    i = j = 0
    while i < n_left and j < n_right:
        lv = left_vals[i]
        rv = right_vals[j]

        if lv == rv:
            # Found a match
            if how in ("inner", "left", "outer"):
                left_indexer[pos] = i
                right_indexer[pos] = j
                pos += 1
                i += 1
            elif how == "right":
                # For right join, find all matches for current left key
                start_i = i
                while j < n_right and right_vals[j] == lv:
                    left_indexer[pos] = i
                    right_indexer[pos] = j
                    pos += 1
                    j += 1
                i = start_i + 1
        elif lv < rv:
            # Left key is smaller - no match in right for this left key
            if how in ("left", "outer"):
                left_indexer[pos] = i
                right_indexer[pos] = -1
                pos += 1
            i += 1
        else:  # lv > rv
            # Right key is smaller - no match in left for this right key
            if how in ("right", "outer"):
                left_indexer[pos] = -1
                right_indexer[pos] = j
                pos += 1
            j += 1

    # Handle remaining elements
    if how in ("left", "outer"):
        while i < n_left:
            left_indexer[pos] = i
            right_indexer[pos] = -1
            pos += 1
            i += 1

    if how in ("right", "outer"):
        while j < n_right:
            left_indexer[pos] = -1
            right_indexer[pos] = j
            pos += 1
            j += 1

    return left_indexer[:pos], right_indexer[:pos]
