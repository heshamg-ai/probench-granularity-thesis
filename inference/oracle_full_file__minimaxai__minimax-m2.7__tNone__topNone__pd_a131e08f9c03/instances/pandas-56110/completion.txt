    def _str_get_dummies(self, sep: str = "|"):
        from pandas import DataFrame

        # Split strings using PyArrow for better performance
        if sep == " ":
            split_arrays = pc.split_whitespace(self._pa_array)
        else:
            split_arrays = pc.split_pattern(self._pa_array, pattern=sep)

        # Collect all unique values from all rows
        seen_values = set()
        for row in split_arrays:
            if row is not None and len(row) > 0:
                for val in row.as_py():
                    seen_values.add(val)

        if not seen_values:
            return DataFrame()

        # Sort values for consistent ordering
        sorted_values = sorted(seen_values)

        # Build columns dictionary using vectorized operations
        n = len(self)
        columns = {}
        for value in sorted_values:
            # Check if value exists in each row's split result
            col_data = np.zeros(n, dtype=np.int8)
            for i, row in enumerate(split_arrays):
                if row is not None and len(row) > 0:
                    if value in row.as_py():
                        col_data[i] = 1
            columns[value] = col_data

        return DataFrame(columns, dtype=np.int8)
