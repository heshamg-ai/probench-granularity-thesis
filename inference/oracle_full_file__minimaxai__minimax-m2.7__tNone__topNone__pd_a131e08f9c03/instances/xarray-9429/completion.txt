1. `UniqueGrouper.group_as_index` - repeated conversion to pd.Index for cftime arrays
2. `unique_value_groups` - using `pd.factorize` which has overhead for cftime objects

    def _factorize_unique(self) -> EncodedGroups:
        # look through group to find the unique values
        index = self.group_as_index
        is_multi = isinstance(index, pd.MultiIndex)
        sort = not is_multi

        if is_multi:
            unique_values, codes_ = pd.factorize(index, sort=False)
            index.names = self.group.name  # type: ignore[union-attr]
        else:
            # Use numpy's unique for better performance with cftime objects
            # np.unique is faster than pd.factorize for arrays of cftime objects
            unique_values, codes_ = np.unique(index, return_inverse=True)

        if (codes_ == -1).all():
            raise ValueError(
                "Failed to group data. Are you grouping by a variable that is all NaN?"
            )

            codes=codes,
            full_index=full_index,

def unique_value_groups(
    ar, sort: bool = True
) -> tuple[np.ndarray | pd.Index, np.ndarray]:
    """Group an array by its unique values.

    Parameters
    ----------
    ar : array-like
        Input array. This will be flattened if it is not already 1-D.
    sort : bool, default: True
        Whether or not to sort unique values.

    Returns
    -------
    values : np.ndarray
        Sorted, unique values as returned by `np.unique`.
    indices : list of lists of int
        Each element provides the integer indices in `ar` with values given by
        the corresponding value in `unique_values`.
    """
    if isinstance(ar, pd.MultiIndex):
        inverse, values = pd.factorize(ar, sort=sort)
        values.names = ar.names
    else:
        # Use numpy's unique for better performance with cftime objects
        # np.unique is faster than pd.factorize for arrays of cftime objects
        if sort:
            unique_values, inverse = np.unique(ar, return_inverse=True)
        else:
            # When sort=False, simulate pd.factorize behavior: preserve first-seen order
            _, indices = np.unique(ar, return_index=True)
            sorted_indices = np.sort(indices)
            inverse = np.empty_like(ar, dtype=np.intp)
            inverse[sorted_indices] = np.arange(len(sorted_indices))
            unique_values = ar[sorted_indices]
        values = unique_values
    return values, inverse
