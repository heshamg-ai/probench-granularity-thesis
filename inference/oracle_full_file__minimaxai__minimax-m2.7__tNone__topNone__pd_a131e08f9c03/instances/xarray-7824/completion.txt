    # Pre-compute cumulative sums for variable indexing (optimization)

    for length in concat_dim_lengths:
        concat_dim_cumsum.append(concat_dim_cumsum[-1] + length)

    # Pre-compute attrs lists for variables that will be concatenated
    # This avoids repeated list comprehensions in the loop
    concat_vars_attrs: dict[Hashable, list] = {}

    # stack up each variable and/or index to fill-out the dataset (in order)

        if name in concat_over and name not in result_indexes:
