        idx_first_valid : type of index

        values = self._values
        ndim = values.ndim if hasattr(values, "ndim") else 2

        if ndim == 2:
            # Optimization for 2D DataFrames with ExtensionArray columns
            # Use each column's native first_valid_index/last_valid_index method
            # which is more efficient than computing a full boolean mask

                for col in values.T:
