class NumericBlock(NumpyBlock):
    __slots__ = ()
    is_numeric = True

    def round(self, decimals: int) -> list[Block]:
        """
        Implement block-level round operation for numeric data.
        """
        if decimals == 0:
            # Fast path: round to nearest integer
            new_values = np.round(self.values, decimals)
        else:
            new_values = np.round(self.values, decimals)
        return [self.make_block(new_values)]

                return ser.round(decimals)

        nv.validate_round(args, kwargs)

        if isinstance(decimals, (dict, Series)):
            if isinstance(decimals, Series) and not decimals.index.is_unique:
                raise ValueError("Index of decimals must be unique")
            if is_dict_like(decimals) and not all(
                is_integer(value) for _, value in decimals.items()
            ):
                raise TypeError("Values in decimals must be integers")
            new_cols = list(_dict_round(self, decimals))
        elif is_integer(decimals):
            # Fast path: use block-level round for numeric columns
            if self._mgr.is_numeric_mixed_type:
                # Mixed numeric/non-numeric, fall back to per-column
                new_cols = [_series_round(v, decimals) for _, v in self.items()]

                # All numeric or all same type - use block-level round
                mgr = self._mgr.round(decimals)
                return self._constructor(mgr).__finalize__(self, method="round")
