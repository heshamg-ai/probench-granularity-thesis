    def _idxmax_idxmin_for_1d(
        self,
        values: ArrayLike,
        ids: np.ndarray,
        ngroups: int,
        how: str,
        skipna: bool,
    ) -> np.ndarray:
        """
        Compute idxmax/idxmin for a 1D array grouped by ids.
        Returns an array of index labels for each group.
        """
        obj_index = self.obj.index

        # Convert to numpy array for processing
        if hasattr(values, "to_numpy"):
            values_arr = values.to_numpy()
        else:
            values_arr = np.asarray(values)

        # Check if we can use the fast numpy path
        dtype = getattr(values_arr, "dtype", None)
        is_numeric_like = (
            is_numeric_dtype(dtype)
            or is_bool_dtype(dtype)
            or is_datetime64_any_dtype(dtype)
            or is_timedelta64_dtype(dtype)
        )

        if not is_numeric_like:
            return self._idxmax_idxmin_for_1d_python(values, ids, ngroups, how, skipna)

        # Handle NA values
        mask = isna(values_arr)
        has_na = mask.any()

        if not skipna and has_na:
            return self._idxmax_idxmin_for_1d_python(values, ids, ngroups, how, skipna)

        # Filter out NA if needed
        if skipna and has_na:
            valid_mask = ~mask
            if not valid_mask.any():
                # All NA
                result = np.empty(ngroups, dtype=object)
                result.fill(np.nan)
                return result
            valid_values = values_arr[valid_mask]
            valid_ids = ids[valid_mask]
            valid_positions = np.nonzero(valid_mask)[0]
        else:
            valid_values = values_arr
            valid_ids = ids
            valid_positions = np.arange(len(values_arr))

        # Filter out dropped groups (ids == -1)
        if (valid_ids == -1).any():
            keep = valid_ids != -1
            valid_values = valid_values[keep]
            valid_ids = valid_ids[keep]
            valid_positions = valid_positions[keep]

        if len(valid_values) == 0:
            result = np.empty(ngroups, dtype=object)
            result.fill(np.nan)
            return result

        # Fast path: use argsort to find min/max per group
        # Sort by group id, then by value
        if how == "idxmax":
            # For max, we sort descending by value, but ascending by position for ties
            # Use negative values for descending sort if numeric
            if is_numeric_dtype(dtype) or is_bool_dtype(dtype):
                # For numeric, negate to find max
                # Handle unsigned by converting to signed
                if dtype.kind == "b":
                    sort_values = -valid_values.astype(np.int8)
                elif dtype.kind == "u":
                    # Unsigned: can't negate, use separate approach
                    _max = valid_values.max() if len(valid_values) > 0 else 0
                    sort_values = _max - valid_values
                else:
                    sort_values = -valid_values
            else:
                # Datetime-like: use view as int64 and negate
                sort_values = -valid_values.view("int64")
        else:
            # For min, sort ascending by value
            sort_values = valid_values

        # argsort by (group_id, sort_value) to bring same groups together
        # and within each group, sort by the value
        # Use stable sort to preserve order for ties
        order = np.lexsort((valid_positions, sort_values, valid_ids))
        # Actually lexsort uses keys in reverse order: sorts by last key first
        # We want: sort by valid_ids first, then by sort_values, then by valid_positions
        # So: lexsort((valid_positions, sort_values, valid_ids)) does:
        #   sort by valid_ids, then sort_values, then valid_positions

        sorted_ids = valid_ids[order]
        sorted_positions = valid_positions[order]

        # Find the first and last occurrence of each unique group in sorted order
        # For each group, if how=="idxmin", the first element in the group's
        # range has the minimum value (due to sort order and stable sort)
        # If how=="idxmax", same logic applies because we negated values

        # Find group boundaries
        if len(sorted_ids) == 0:
            result = np.empty(ngroups, dtype=object)
            result.fill(np.nan)
            return result

        # Find unique group ids and their positions
        # We need to handle all ngroups, including empty ones
        result = np.empty(ngroups, dtype=object)
        result.fill(np.nan)

        # Use searchsorted to find group boundaries efficiently
        # Find where each group starts and ends in sorted_ids

        # Get the observed group ids
        unique_groups, start_indices = np.unique(sorted_ids, return_index=True)

        # For each unique group, get the relevant element
        for i, grp in enumerate(unique_groups):
            start = start_indices[i]
            end = start_indices[i + 1] if i + 1 < len(start_indices) else len(sorted_ids)
            # Due to our sort order, start gives the desired element
            # For min: first element (smallest value, earliest position for ties)
            # For max: first element after sorting by negated value
            result_pos = sorted_positions[start]
            result[grp] = obj_index[result_pos]

            # But we need to verify: for max with negation, does the first element
            # in the group have the maximum value? Yes, because we sort by negated
            # value ascending, so the first has most negative = largest original.

        return result

        result = np.empty(ngroups, dtype=object)

        from collections import defaultdict

                else:

                pass

        return result
