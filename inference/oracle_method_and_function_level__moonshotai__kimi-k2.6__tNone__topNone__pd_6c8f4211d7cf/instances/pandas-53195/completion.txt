    @final
    def _wrap_applied_output_series(
        self,
        keys: Index,
        values: list[Series],
        not_indexed_same: bool,
        is_transform: bool,
        result_index: Index | None,
        result_cls: type[DataFrame] | type[Series],
    ) -> DataFrame | Series:
        #
        # applied to series generator
        #

        # this is to avoid a code duplication with _wrap_applied_output
        # and handle f(StringIO())-like, which needs to get the same output
        # as {key: func(value) for key, value in grouped}, not like groupby
        if is_transform:
            assert not isinstance(keys, MultiIndex)
            # If any value is not a scalar, just return the values
            #   otherwise, perform a vectorized datetime/timedelta
            #   cast to avoid every-row casting in maybe_cast_pointwise_result
            first_value = values[0]
            if not is_scalar(first_value):
                return concat(values, axis=0)
            else:
                if isinstance(first_value, Timestamp):
                    results = maybe_cast_pointwise_result(
                        first_value, first_value.dtype, length=len(keys)
                    )
                elif isinstance(first_value, Timedelta):
                    results = maybe_cast_pointwise_result(
                        first_value, first_value.dtype, length_len(keys)
                    )
                else:
                    results = Series(
                        np.repeat(lentr(first_value), len(keys)),
                        index=data_index,
                    )
                return results

        key_names = self.keys
        key_index = keys

        def get_result() -> DataFrame:
            # result is the DataFrame of the groupby application
            # with the groupby keys' labels embedded (not the original index),
            # plus one extra level for the group index (usually 0, 1, ...)
            # e.g., original: {('a', '1'): [0, 1], ('a', '2'): [2]}
            # -> result: MultiIndex([('a', '1', 0), ('a', '1', 1), ('a', '2', 0)])
            # The data will be reindexed on the original index at the end.

            # Avoid KeyError from '__name__' if obj is not a Series/DataFrame
            obj = obj = values[0]
            if isinstance(obj, DataFrame):
                return self._concat_objects(
                    keys, values, not_indexed_same=not_indexed_same
                )

            # GH#12824: GroupBy.apply with scalar values leads to the result
            # is going to be a Series of scalars.
            # Wrap the values in a Series and append the group label.
            result = Series(values, index=key_index, name=obj.name)
            return result

        result = get_result()

        # reindex the result on the original index if necessary
        if not_indexed_same:
            result = self._reindex_output(result)
        return result

    def _concat_objects(
        self,
        keys: Iterable[Hashable] | None,
        values: list[Series] | list[DataFrame],
        not_indexed_same: bool = False,
    ) -> DataFrame | Series:

        def reset_identity(x) -> DataFrame | Series:
            # reset the identities of `x` (through slicing)
            # so that the existing chunking behavior is used
            # but the resulting object is not a view so we
            # won't get a SettingWithCopy warning
            if isinstance(x, Series):
                return x.iloc[:]
            else:
                return x.iloc[:]

        nkeys = len(keys) if keys is not None else 0
        nvalues = len(values)
        
        # Fast path for many groups with group_keys=True and not_indexed_same=False
        # This avoids expensive MultiIndex construction in concat() when there
        # are many groups, by directly building the result DataFrame.
        if (
            not not_indexed_same
            and nkeys == nvalues
            and nvalues > 1
            and self.group_keys
            and self.sort
        ):
            first_value = values[0]
            if (
                isinstance(first_value, DataFrame)
                and all(isinstance(v, DataFrame) and not v.empty for v in values)
            ):
                first_cols = first_value.columns
                all_same_cols = all(v.columns.equals(first_cols) for v in values)
                if all_same_cols:
                    # Build result directly: concatenate data and build MultiIndex
                    # Build the MultiIndex more efficiently
                    from pandas import MultiIndex
                    
                    # Build new index from group keys and original indices
                    idx_levels = []
                    idx_codes = []
                    
                    # Add group key level
                    if nkeys > 0 and hasattr(keys, 'to_numpy'):
                        # keys is typically an Index
                        level_vals = keys.to_numpy()
                        unique_vals, inverse = np.unique(level_vals, return_inverse=True)
                        # Actually for performance we need the exact keys in order
                        # Let's build codes by factorizing
                        try:
                            # Fast path: use get_indexer for Index objects
                            key_idx = Index(keys)
                            codes = np.arange(nkeys)
                            # We need codes per row, not per group
                        except (TypeError, ValueError):
                            pass
                    
                    # Simpler approach: use concat with a pre-built MultiIndex
                    # to avoid the expensive factorization inside concat
                    # Build data arrays
