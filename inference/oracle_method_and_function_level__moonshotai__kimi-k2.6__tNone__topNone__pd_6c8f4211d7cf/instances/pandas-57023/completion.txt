        ridx: npt.NDArray[np.intp],
        how: JoinHow,

            join_idx = self.take(lidx)
            right = cast("MultiIndex", other.take(ridx))
            join_index = join_idx.putmask(mask, right)._sort_levels_monotonic()
            return join_index.set_names(names)  # type: ignore[return-value]

        else:
            result = self._constructor._with_infer(
                joined, name=names[0], dtype=self.dtype
            )
            # Reuse cached attributes from self or other when the result
            # matches one of the input indexes exactly.
            if how == "left" and lidx is None and ridx is not None:
                # result is identical to self
                if hasattr(self, "_cache"):
                    for key, value in self._cache.items():
                        result._cache[key] = value
                result._cache.update({k: v for k, v in self._cache.items()})
            elif how == "right" and ridx is None and lidx is not None:
                # result is identical to other
                if hasattr(other, "_cache"):
                    for key, value in other._cache.items():
                        result._cache[key] = value
            elif how in ("inner", "outer") and lidx is not None and ridx is not None:
                # Check if the result is identical to self or other

                    len(lidx) == len(self) and (lidx == np.arange(len(self))).all()

                ):

                        for key, value in other._cache.items():
                            result._cache[key] = value
            return result
