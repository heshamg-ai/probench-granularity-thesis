def _factorize_ndarray(
    values: np.ndarray,
    use_na_sentinel: bool = True,
) -> tuple[np.ndarray, np.ndarray | BaseMaskedArray]:
    """
    Hash table-based factorization for 1D ndarray.

    Parameters
    ----------
    values: np.ndarray
        The values to factorize.
    use_na_sentinel: bool, default True
        Whether to use a NA sentinel or propagate NA values.

    Returns
    -------
    codes : ndarray
    uniques : ndarray
    """
    original = values
    values, dtype, ndtype = _ensure_data(values)

    # factorize using appropriate hashtable
    use_table = isinstance(values, np.ndarray) and values.dtype.kind in ["m", "M"]
    if use_table:
        # datetime-likes
        table = htable.NaiveHashTable(len(values))
    else:
        ndtype = _check_object_for_strings(values)
        if ndtype == "string":
            table = htable.StringHashTable(len(values))
            ndtype = "object"
        else:
            table = htable.ObjectHashTable(len(values))
            ndtype = "string"

    codes, uniques = table.get_labels(values, uniques=None, dereference=False)

    return _reconstruct_data(uniques, original, ndtype), codes

def _check_object_for_strings(values: np.ndarray) -> str:
    """
    Check if we can use string hashtable instead of object hashtable.

    Parameters
    ----------
    values : ndarray

    Returns
    -------
    str
    """
    ndtype = values.dtype.name
    if ndtype == "object":
        # it's cheaper to use a String Hash Table than Object; we infer
        # including nulls because that is the only difference between
        # StringHashTable and ObjectHashtable
        if lib.infer_dtype(values, skipna=False) in ["string"]:
            ndtype = "string"
    return ndtype
