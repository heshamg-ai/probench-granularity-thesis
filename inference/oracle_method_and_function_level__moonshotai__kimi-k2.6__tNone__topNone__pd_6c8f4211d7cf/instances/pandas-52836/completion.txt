def transpose_homogenous_masked_arrays(
    dtype: ExtensionDtype,
    values: list[BaseMaskedArray],
) -> list[BaseMaskedArray]:
    """
    Fast transpose for homogeneous masked arrays.

    Parameters
    ----------
    dtype : ExtensionDtype
        The common dtype of all arrays.
    values : list[BaseMaskedArray]
        List of masked arrays (columns) to transpose.

    Returns
    -------
    list[BaseMaskedArray]
        List of masked arrays representing the transposed rows.
    """
    n_rows = len(values[0])
    n_cols = len(values)

    # Stack data and masks into 2D numpy arrays
    stacked_data = np.vstack([arr._data for arr in values])
    stacked_mask = np.vstack([arr._mask for arr in values])

    # Transpose
    transposed_data = stacked_data.T
    transposed_mask = stacked_mask.T

    # Construct result arrays
    arr_type = dtype.construct_array_type()
    result = [
        arr_type(transposed_data[i], transposed_mask[i])  # type: ignore[call-arg]
        for i in range(n_rows)
    ]

    return result
