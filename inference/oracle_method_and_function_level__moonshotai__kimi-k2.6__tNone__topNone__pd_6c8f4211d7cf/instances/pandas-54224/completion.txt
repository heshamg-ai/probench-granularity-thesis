def transpose_homogeneous_pyarrow(
    arrays: list[ArrowExtensionArray],
) -> list[ArrowExtensionArray]:
    """Transpose PyArrow-backed arrays in a list, but faster.

    Input should be a list of 1-dim ArrowExtensionArray of equal length and all
    have the same dtype. The caller is responsible for ensuring validity of
    input data.
    """
    import pyarrow as pa

    dtype = arrays[0].dtype
    arrow_type = dtype.pyarrow_dtype

    # Use ChunkedArray to avoid building intermediate Python lists
    # Concatenate all arrays into a single ChunkedArray
    chunks = []
    for arr in arrays:
        # Extract the pyarrow array from each ArrowExtensionArray
        pa_arr = arr._pa_array
        chunks.append(pa_arr)

    # Create a ChunkedArray from all chunks
    chunked = pa.chunked_array(chunks)
    # Combine into a single array for easier manipulation
    combined = chunked.combine_chunks()

    # Now we have a single list of values, we need to reshape them
    # There are `len(arrays)` arrays each of length `len(arrays[0])`
    n_rows = len(arrays)
    n_cols = len(arrays[0])

    # Reshape by taking strides
    result_arrays = []
    for i in range(n_cols):
        # Extract every n_rows-th element starting from i
        # This gives us the i-th row of the transposed matrix
        row = combined.slice(i, n_rows * n_rows - i)
        # Use take with indices to get strided elements
        indices = pa.array(range(i, n_rows * n_cols, n_rows))
        row_taken = combined.take(indices)
        result_arrays.append(ArrowExtensionArray(row_taken))

    return result_arrays
