def _round_through_cftimeindex(values, name, freq):
    """Coerce an array of datetime-like values to a CFTimeIndex
    and apply the requested rounding operation (floor/ceil/round)."""
    from xarray.coding.cftimeindex import CFTimeIndex

    if is_duck_dask_array(values):
        from dask.array import map_blocks

        return map_blocks(
            _round_through_cftimeindex, values, name, freq=freq, dtype=values.dtype
        )

    values_as_cftimeindex = CFTimeIndex(values.ravel())
    method = getattr(values_as_cftimeindex, name)
    rounded = method(freq)
    return rounded.values.reshape(values.shape)
