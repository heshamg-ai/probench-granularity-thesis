def _multinomial_loss_label(w, X, y, alpha, sample_weight, n_classes):
    """Computes multinomial loss and class probabilities using integer labels.

    Parameters
    ----------
    w : ndarray of shape (n_classes * n_features,) or
        (n_classes * (n_features + 1),)
        Coefficient vector.

    X : {array-like, sparse matrix} of shape (n_samples, n_features)
        Training data.

    y : ndarray of shape (n_samples,)
        Integer-encoded target labels in [0, n_classes).

    alpha : float
        Regularization parameter. alpha is equal to 1 / C.

    sample_weight : array-like of shape (n_samples,)
        Array of weights that are assigned to individual samples.

    n_classes : int
        Number of classes.

    Returns
    -------
    loss : float
        Multinomial loss.

    p : ndarray of shape (n_samples, n_classes)
        Estimated class probabilities.

    w_reshaped : ndarray of shape (n_classes, n_features)
        Reshaped param vector excluding intercept terms.
    """
    n_features = X.shape[1]
    n_classes = int(n_classes)
    fit_intercept = w.size == (n_classes * (n_features + 1))
    w = w.reshape(n_classes, -1)
    if fit_intercept:
        intercept = w[:, -1]
        w = w[:, :-1]
    else:
        intercept = 0
    p = safe_sparse_dot(X, w.T)
    p += intercept
    # compute log probabilities in-place
    log_sum = logsumexp(p, axis=1)
    p -= log_sum[:, np.newaxis]
    # negative log-likelihood using integer labels
    loss = -(sample_weight * p[np.arange(X.shape[0]), y]).sum()
    loss += 0.5 * alpha * squared_norm(w)
    # convert to probabilities in-place
    np.exp(p, out=p)
    return loss, p, w

def _multinomial_loss_grad_label(w, X, y, alpha, sample_weight, n_classes):
    """Computes multinomial loss, gradient and class probabilities using labels."""
    n_features = X.shape[1]
    n_classes = int(n_classes)
    fit_intercept = w.size == n_classes * (n_features + 1)
    loss, p, w_no_intercept = _multinomial_loss_label(
        w, X, y, alpha, sample_weight, n_classes
    )
    grad = np.zeros((n_classes, n_features + bool(fit_intercept)), dtype=X.dtype)
    # subtract 1 on the true class column for each sample
    p[np.arange(X.shape[0]), y] -= 1.0
    # apply sample weights
    p *= sample_weight[:, np.newaxis]
    grad[:, :n_features] = safe_sparse_dot(p.T, X)
    grad[:, :n_features] += alpha * w_no_intercept
    if fit_intercept:
        grad[:, -1] = p.sum(axis=0)
    return loss, grad.ravel(), p
