    # Vectorized binary search along axis=1 to avoid per-feature Python loop
    n_features, n_samples = weight_cdf.shape
    idx_dtype = sorted_idx.dtype
    low = xp.zeros(n_features, dtype=idx_dtype, device=device)
    high = xp.full(n_features, n_samples, dtype=idx_dtype, device=device)
    # Number of iterations to converge for binary search
    iters = (n_samples - 1).bit_length()
    for _ in range(iters):
        mid = (low + high) // 2

        cond = mid_vals < adjusted_percentile_rank
        low = xp.where(cond, mid + 1, low)
