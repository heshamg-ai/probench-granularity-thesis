        from pandas.core.series import Series
        from pandas.core.dtypes.missing import isna
        from pandas.api.types import is_scalar

        columns = ensure_index(columns)
        col_list = list(columns)

        # Determine missing mask and collect present values for index inference
        missing_mask: list[bool] = []
        present_vals = []
        for col in col_list:
            if col in data:
                v = data[col]
                m = bool(isna(v)) if is_scalar(v) else False
                missing_mask.append(m)
                if not m:
                    present_vals.append(v)
            else:
                missing_mask.append(True)

        if index is None:
            # GH10856
            # raise ValueError if only scalars in dict
            idx_source = Series(present_vals, dtype=object)
            index = _extract_index(idx_source)
        else:
            index = ensure_index(index)

        # Build arrays aligned to columns; placeholder np.nan for missing
        arrays = [data[c] if not m else np.nan for c, m in zip(col_list, missing_mask)]

        # no obvious "empty" int column
