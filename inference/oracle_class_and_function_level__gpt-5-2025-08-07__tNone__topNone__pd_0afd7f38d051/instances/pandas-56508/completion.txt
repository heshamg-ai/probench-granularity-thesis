    def _hash_pandas_object(
        self, *, encoding: str, hash_key: str, categorize: bool
    ) -> npt.NDArray[np.uint64]:
        from pandas.core.util.hashing import hash_array

        codes, uniques = self.factorize(use_na_sentinel=True)
        n = len(codes)
        if n == 0:
            return np.empty(0, dtype=np.uint64)

        # Hash unique values once and map by codes; this matches per-element hashing
        # semantics while avoiding materializing object arrays.
        hashed_uniques = hash_array(
            uniques._data, encoding=encoding, hash_key=hash_key, categorize=False
        )

        result = np.empty(n, dtype=np.uint64)
        if len(hashed_uniques) == 0:
            # all values are NA
            result.fill(0)
            return result

        mask_na = codes == -1
        if mask_na.any():
            result[mask_na] = np.uint64(0)
            if (~mask_na).any():
                result[~mask_na] = hashed_uniques[codes[~mask_na]]
        else:
            result[:] = hashed_uniques[codes]
        return result
