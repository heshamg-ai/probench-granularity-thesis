def _round_through_series(values, name, freq):
    """Round np.datetime64 values using pandas Series."""
    import pandas as pd

    series = pd.Series(values.ravel())
    rounded = getattr(series.dt, name)(freq)
    return rounded.values.reshape(values.shape)


def _round_through_cftimeindex(values, name, freq):
    """Round cftime values using CFTimeIndex."""
    from xarray.coding.cftimeindex import CFTimeIndex

    values_as_cftimeindex = CFTimeIndex(values.ravel())
    rounded = getattr(values_as_cftimeindex, name)(freq)
    return rounded.values.reshape(values.shape)
