        unique_values, codes_ = unique_value_groups(self.group_as_index, sort=sort)
        if (codes_ == -1).all():

            )
        codes = self.group.copy(data=codes_.reshape(self.group.shape))

        )

            full_index=full_index,
            unique_coord=unique_coord,

        )
