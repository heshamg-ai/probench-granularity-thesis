    def _groupby_op(
        self,
        *,
        how: str,
        has_dropped_na: bool,
        min_count: int,
        ngroups: int,
        ids: npt.NDArray[np.intp],
        **kwargs,
    ):
        if isinstance(self.dtype, StringDtype):
            return super()._groupby_op(
                how=how,
                has_dropped_na=has_dropped_na,
                min_count=min_count,
                ngroups=ngroups,
                ids=ids,
                **kwargs,
            )

        pa_type = self._pa_array.type

        # Fast path for min/max on temporal and duration types:
        # Use pyarrow compute group-by directly, avoiding cast to masked
        # and the per-group Python overhead.
        if how in ("max", "min") and (
            pa.types.is_timestamp(pa_type) or pa.types.is_duration(pa_type)
        ):
            return self._groupby_op_arrow_min_max(
                how=how,
                ngroups=ngroups,
                ids=ids,
                min_count=min_count,
                has_dropped_na=has_dropped_na,
            )

        masked = self._to_masked()

        result = masked._groupby_op(
            how=how,
            has_dropped_na=has_dropped_na,
            min_count=min_count,
            ngroups=ngroups,
            ids=ids,
            **kwargs,
        )
        if isinstance(result, np.ndarray):
            return result
        return type(self)._from_sequence(result, copy=False)

    def _groupby_op_arrow_min_max(
        self,
        *,
        how: str,
        ngroups: int,
        ids: npt.NDArray[np.intp],
        min_count: int,
        has_dropped_na: bool,
    ) -> np.ndarray | Self:
        """
        Fast min/max groupby aggregation for timestamp and duration types.

        Uses pyarrow's compute group-by functions which are implemented in C++
        and operate directly on the Arrow data without per-group Python overhead.
        """
        pa_type = self._pa_array.type
        data = self._pa_array.combine_chunks()

        # For duration types, pyarrow min/max requires int64
        if pa.types.is_duration(pa_type):
            data = data.cast(pa.int64())

        # Build a pyarrow array of group IDs
        ids_arrow = pa.array(ids, type=pa.int64())

        # Perform the group-by aggregation using pyarrow compute
        if how == "max":
            result = pc.max(data, ids_arrow)
        else:  # min
            result = pc.min(data, ids_arrow)

        # Ensure result has `ngroups` elements
        # If some groups have no observations, pyarrow returns null for them
        if len(result) < ngroups:
            # Extend result to match ngroups (unlikely path, but safe)
            mask = pa.array(
                [False] * len(result) + [True] * (ngroups - len(result)),
                type=pa.bool_(),
            )
            result = pc.if_else(
                mask, pa.scalar(None, type=result.type), result
            )

        # Cast back to original type for duration
        if pa.types.is_duration(pa_type):
            result = result.cast(pa_type)

        # Convert to numpy or ArrowExtensionArray
        if min_count > 0:
            # Need to handle min_count: count non-null per group and mask
            counts = pc.count(data, ids_arrow)
            if len(counts) < ngroups:
                counts_mask = pa.array(
                    [False] * len(counts) + [True] * (ngroups - len(counts)),
                    type=pa.bool_(),
                )
                counts = pc.if_else(
                    counts_mask,
                    pa.scalar(0, type=pa.int64()),
                    counts,
                )
            below_min_count = pc.less(counts, min_count)
            result = pc.if_else(
                below_min_count,
                pa.scalar(None, type=result.type),
                result,
            )

        # Convert nulls to NA
        if has_dropped_na:
            # If groups with all NA were already dropped, we may not need to
            # fill NA — return as numpy with NA values
            result_np = result.to_numpy(zero_copy_only=False)
            return result_np

        # Return as ArrowExtensionArray preserving NA
        return type(self)(result)
