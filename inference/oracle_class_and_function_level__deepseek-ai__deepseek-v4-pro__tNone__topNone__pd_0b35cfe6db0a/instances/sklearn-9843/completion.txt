    sample_weight = sample_weight[ind]

    # Choose the accumulator dtype to always have high precision
    if sample_weight.dtype.kind in {'i', 'u', 'b'}:
        dtype = np.int64
    else:
        dtype = np.float64

    # Use np.bincount + ravel_multi_index when labels are integral and
    # sample_weight is integral (the common case), which is faster and
    # uses less memory than coo_matrix.
    if sample_weight.dtype.kind in {'i', 'u', 'b'}:
        cm = np.zeros((n_labels, n_labels), dtype=dtype)
        np.add.at(cm, (y_true, y_pred), sample_weight)
    else:
