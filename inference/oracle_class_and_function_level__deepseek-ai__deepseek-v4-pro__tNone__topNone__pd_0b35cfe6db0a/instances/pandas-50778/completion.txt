    @staticmethod
    def _replace_list_fast(
        values: np.ndarray,
        pairs: list[tuple[Any, Any]],
        masks: list[npt.NDArray[np.bool_]],
        any_mask: npt.NDArray[np.bool_],
    ) -> None:
        """
        Apply a list of replacements in a single vectorized pass
        when all destination values can be held by the array.

        Parameters
        ----------
        values : np.ndarray
            Values to modify in-place.
        pairs : list of tuple
            (src, dest) pairs for replacement.
        masks : list of np.ndarray[bool]
            Boolean masks corresponding to each pair, same shape as values.
        any_mask : np.ndarray[bool]
            Pre-computed combined mask (OR of all individual masks).
        """
        if values.dtype.kind in ("i", "u", "f", "c", "b"):
            # For numeric types, build a reverse-map array and use
            # direct indexing.  This is effectively a dict-based
            # replacement in one pass.
            # Determine if all src values fit within a reasonable range
            # for array indexing.  If so, build a mapping array.
            src_values = [s for s, _ in pairs]
            dst_values = [d for _, d in pairs]
            if all(isinstance(s, (int, np.integer)) for s in src_values):
                src_ints = [int(s) for s in src_values]
                max_src = max(src_ints)
                if max_src < 10_000_000:
                    # Build a flat replacement array.
                    # We don't know the exact dtype of src/dst, so use
                    # the array's dtype to initialize.
                    dt = values.dtype
                    # For safety, build a mapping array of dtype `dt`
                    # initialized with a sentinel (the identity mapping).
                    # This only works when dst values are directly
                    # representable in `dt`.
                    map_arr = np.arange(max_src + 1, dtype=dt)
                    for s, d in zip(src_ints, dst_values):
                        if isinstance(d, float) and dt.kind != "f":
                            # Casting float to int would lose precision;
                            # fall back to the mask-based approach below.
                            break
                        map_arr[s] = dt.type(d)
                    else:
                        # All destinations fit; apply via indexing.
                        flat_values = values.ravel()
                        flat_mask = any_mask.ravel()
                        idx = flat_values[flat_mask]
                        # Clip to avoid index errors for values > max_src
                        valid = idx <= max_src
                        idx_valid = idx[valid]
                        flat_values[flat_mask] = np.where(
                            valid,
                            map_arr[idx_valid],
                            flat_values[flat_mask],
                        )
                        return

        # Fallback: apply masks in order without splitting blocks.
        # This still avoids the overhead of block splitting and
        # repeated coercion checks.
        for pair_mask, (_, dest) in zip(masks, pairs):
            putmask_inplace(values, pair_mask, dest)
