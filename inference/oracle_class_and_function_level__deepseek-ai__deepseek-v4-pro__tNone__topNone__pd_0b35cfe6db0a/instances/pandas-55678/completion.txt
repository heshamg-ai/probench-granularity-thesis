def _convert_by_keys_to_common_dtype(
    lbv: np.ndarray, rbv: np.ndarray
) -> tuple[np.ndarray, np.ndarray]:
    """
    Convert left and right 'by' keys to a common dtype when they differ,
    avoiding unnecessary copies when dtypes already match.
    """
    # Already handled the fully-matching case in the single-key fast path;
    # this function is called when they differ or when we can't determine.
    if not isinstance(lbv, np.ndarray):
        lbv = np.asarray(lbv)
    if not isinstance(rbv, np.ndarray):
        rbv = np.asarray(rbv)

    if lbv.dtype == rbv.dtype:
        if needs_i8_conversion(lbv.dtype):
            lbv = lbv.view("i8")
            rbv = rbv.view("i8")
        return lbv, rbv

    # Determine a common dtype that is safe and compatible
    if needs_i8_conversion(lbv.dtype) and needs_i8_conversion(rbv.dtype):
        # Both are datetime-like; convert to their i8 views
        lbv = lbv.view("i8")
        rbv = rbv.view("i8")
        return lbv, rbv

    # Fallback: find common NumPy dtype via result_type
    common_dtype = np.result_type(lbv, rbv)
    if lbv.dtype != common_dtype:
        lbv = lbv.astype(common_dtype, copy=False)
    if rbv.dtype != common_dtype:
        rbv = rbv.astype(common_dtype, copy=False)
    return lbv, rbv


def _factorize_by_keys(
    left_by_values: list[ArrayLike],
    right_by_values: list[ArrayLike],
) -> list[tuple[np.ndarray, np.ndarray, int]]:
    """
    Factorize multiple 'by' key columns, converting to common dtypes
    where possible to avoid repeated work in _factorize_keys.
    """
    results: list[tuple[np.ndarray, np.ndarray, int]] = []
    for n in range(len(left_by_values)):
        lv = left_by_values[n]
        rv = right_by_values[n]
        if not isinstance(lv, np.ndarray):
            lv = np.asarray(lv)
        if not isinstance(rv, np.ndarray):
            rv = np.asarray(rv)

        # If dtypes differ, convert to common dtype once to save
        # _factorize_keys from performing this internally
        if lv.dtype != rv.dtype:
            lv, rv = _convert_by_keys_to_common_dtype(lv, rv)

        mapped = _factorize_keys(
            lv,
            rv,
            sort=False,
            how="left",
        )
        results.append(mapped)
    return results
