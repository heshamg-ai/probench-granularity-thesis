def _get_group_names(regex: re.Pattern) -> list[Hashable]:
    """
    Get named groups from compiled regex.

    Unnamed groups are numbered.

    Parameters
    ----------
    regex : compiled regex

    Returns
    -------
    list of column labels
    """
    num_groups = regex.groups
    groupindex = regex.groupindex
    if not groupindex:
        # Fast path: no named groups, use RangeIndex via a sentinel
        return list(range(num_groups))
    names = {v: k for k, v in groupindex.items()}
    result = [None] * num_groups
    for i in range(num_groups):
        key = i + 1
        if key in names:
            result[i] = names[key]
    return result
