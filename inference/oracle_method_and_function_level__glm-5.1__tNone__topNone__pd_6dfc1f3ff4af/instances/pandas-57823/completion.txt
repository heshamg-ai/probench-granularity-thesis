    def argmin(self, axis=None):
        self._validate_axis(axis)
        if not self.is_monotonic_increasing_or_decreasing:
            return super().argmin(axis=axis)
        if len(self) == 0:
            raise ValueError("attempt to get argmin of an empty sequence")
        if self.step > 0:
            return 0
        return len(self) - 1
