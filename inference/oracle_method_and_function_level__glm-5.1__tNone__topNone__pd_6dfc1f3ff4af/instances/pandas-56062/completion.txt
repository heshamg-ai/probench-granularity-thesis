    def _getitem_tuple(self, tup: tuple):
        # Optimization: fast path for MultiIndex scalar lookups
        # Avoids expensive general-purpose tuple indexing path
        if (
            len(tup) == self.ndim
            and self.obj.index._is_multi
            and all(is_scalar(x) for x in tup)
        ):
            # Attempt direct hash-based lookup for each dimension
            try:
                indexer = self.obj.index.get_loc(tup)
                if isinstance(indexer, int):
                    return self.obj._ixs(indexer, axis=0)
            except (KeyError, TypeError):
                pass  # Fall through to standard path

        try:
            return self._getitem_tuple_same_dim(tup)
        except IndexingError:
            return self._getitem_tuple_diff_dim(tup)
