def _map_int_labels(y, label_to_ind, n_labels):
    """Optimized label mapping for integer arrays using vectorized numpy ops."""
    y = np.asarray(y)
    y_min = y.min()
    y_max = y.max()
    # size of lookup table is bounded by the range of the data
    range_size = int(y_max) - int(y_min) + 1
    if range_size <= 2 * y.size:
        lut = np.full(range_size, n_labels + 1, dtype=np.intp)
        for label, idx in label_to_ind.items():
            if y_min <= label <= y_max:
                lut[label - y_min] = idx
        y_out = lut[y - y_min]
    else:
        y_out = np.empty(y.shape, dtype=np.intp)
        y_out[:] = n_labels + 1
        for label, idx in label_to_ind.items():
            y_out[y == label] = idx
    return y_out
