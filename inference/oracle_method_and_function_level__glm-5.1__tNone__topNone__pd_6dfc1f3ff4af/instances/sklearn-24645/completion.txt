def _sqeuclidean_row_norms(X):
    """Compute the squared Euclidean norm for each row of X."""
    if sp.issparse(X):
        return np.asarray(X.multiply(X).sum(axis=1)).ravel()
    return np.einsum("ij,ij->i", X, X)
