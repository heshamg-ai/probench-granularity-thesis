        # Optimization: for non-regex, non-categorical, non-extension,
        # non-object blocks, use vectorized searchsorted on unique values
        # to compute all masks at once.
        if (
            not regex
            and not isinstance(values, Categorical)
            and not self.is_extension
            and not self.is_object
            and not is_string_dtype(values.dtype)
            and not isna(values).any()
        ):
            uniques, inv = algorithms.unique(values, return_inverse=True)
            src_arr = np.array(
                [s[0] for s in pairs], dtype=values.dtype if hasattr(values, 'dtype') else None
            )
            idx = uniques.searchsorted(src_arr, side='left')
            idx[idx == len(uniques)] = -1
            found = uniques[idx] == src_arr

            masks = []
            for i in range(len(pairs)):
                m = np.zeros(values.shape, dtype=np.bool_)
                if found[i]:
                    m[inv == idx[i]] = True
                masks.append(m)
        elif is_string_dtype(values.dtype):
