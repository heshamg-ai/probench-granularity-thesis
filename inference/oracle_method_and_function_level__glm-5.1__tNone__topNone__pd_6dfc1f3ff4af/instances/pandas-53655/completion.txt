    def _str_get_dummies(self, sep: str = "|"):
        split = pc.split_pattern(self._pa_array, sep).combine_chunks()
        uniques = split.flatten().unique()
        uniques_sorted = uniques.take(pa.compute.array_sort_indices(uniques))
        n_unique = len(uniques_sorted)
        unique_set = set(uniques_sorted.to_pylist())
        indices = pc.index_in(uniques_sorted, pa.array(sorted(unique_set)))
        result_data = []
        for lst in split.to_pylist():
            if lst is None:
                result_data.append([False] * n_unique)
            else:
                row = [False] * n_unique
                for v in lst:
                    idx = indices[v].as_py()
                    if idx is not None:
                        row[idx] = True
                result_data.append(row)
        result = type(self)(pa.array(result_data))
        return result, uniques_sorted.to_pylist()
